    const safeJson = (data) => {
        if (!data) return [];
        let arr = [];
        if (Array.isArray(data)) {
            arr = data;
        } else if (typeof data === 'object') {
            arr = Object.entries(data)
                        .filter(([key]) => !key.startsWith('$'))
                        .map(([, value]) => value);
        } else if (typeof data === 'string') {
            arr = [data];
        }
        return arr.filter(t => t && t !== '$__META_EXTENSIBLE__$');
    };
    
    const updateDOM = (f, t) => { 
        if (!f || !t || f.isEqualNode(t)) return; 
        if (f.nodeType === 3) {
            if (f.textContent !== t.textContent) f.textContent = t.textContent;
            return;
        } 
        if (f.nodeType !== 1) return; 
        Array.from(t.attributes).forEach(a => f.setAttribute(a.name, a.value)); 
        Array.from(f.attributes).forEach(a => {
            if (!t.hasAttribute(a.name)) f.removeAttribute(a.name);
        }); 
        const fl = f.childNodes.length, tl = t.childNodes.length; 
        for (let i = 0; i < Math.max(fl, tl); i++) { 
            if (i >= fl) f.appendChild(t.childNodes[i].cloneNode(true)); 
            else if (i >= tl) f.removeChild(f.childNodes[i]); 
            else updateDOM(f.childNodes[i], t.childNodes[i]); 
        } 
    };

    // 基础条
    const renderBar = (l, i, v, m, c, k, extraCls='', isPregnant=false) => {
        let fetalNotchHtml = '';
        let fetalLabelHtml = '';
        if (k === 'hp' && isPregnant) {
            const fetalPct = Math.min(100, (20 / m) * 100);
            fetalNotchHtml = `<div class="fetal-hp-notch-mob" style="width:${fetalPct}%;" onclick="openModal('fetal_hp')" title="胎儿汲取占用 20 HP"></div>`;
            fetalLabelHtml = `<span style="font-size:0.75em; color:var(--pleasure-color); font-weight:normal; margin-left:4px;" onclick="openModal('fetal_hp')"><i class="fa-solid fa-baby"></i> -20</span>`;
        }
        return `
        <div class="bar-group">
            <div class="bar-label-row">
                <span><i class="${i}"></i> ${l} <i class="fa-solid fa-circle-info info-icon" onclick="openModal('${k}')"></i>${fetalLabelHtml}</span>
                <span class="bar-value">${Math.round(v)}</span>
            </div>
            <div class="bar-track" style="position:relative;">
                <div class="bar-fill ${extraCls}" style="width:${Math.min(100, (v/m)*100)}%; background:var(${c}); position:relative; z-index:1;"></div>
                ${fetalNotchHtml}
            </div>
        </div>`;
    };
    
    // 战斗面板渲染
    const renderCombat = (data, phase) => {
        const hp = getValue(data, '世界.当前关卡.Boss韧性值', 100);
        const name = getValue(data, '世界.当前关卡.名称', 'Unknown');
        const threat = getValue(data, '世界.威胁等级', 1);

        const cType = getValue(data, '世界.当前关卡.战斗类型', '未知');

        let typeIcon = 'fa-spider';
        if (cType === '魔法少女') typeIcon = 'fa-moon';
        else if (cType === '人类') typeIcon = 'fa-user-secret';
        
        let statusHtml = '';
        let alertHtml = ''; 
        let borderColor = 'var(--combat-color)';
        
        if (phase === '战斗准备') {
            borderColor = 'var(--ultimate-color)';

                else { alertColor = 'var(--hp-color)'; alertLabel = '隐蔽/安全'; }

                alertHtml = `

                    <div style="display:flex; justify-content:space-between; font-size:0.85em; color:${alertColor}; font-weight:bold; margin-bottom:5px; align-items:center; white-space:nowrap; gap:6px;">
                        <span onclick="openModal('alertness')" style="cursor:pointer; display:inline-flex; align-items:center; gap:4px; min-width:0; overflow:hidden; text-overflow:ellipsis;">
                            <i class="fa-solid fa-eye"></i> 警戒: [${alertLabel}] <i class="fa-solid fa-circle-info info-icon" style="font-size:0.8em; opacity:0.6;"></i>

                            <span style="font-size:0.8em; opacity:0.8; font-weight:normal; margin-right:3px;">(+${alertness} 敌骰)</span>${alertness}<span style="font-size:0.8em; opacity:0.6; font-weight:normal;">/10</span>
                        </span>
                    </div>
                    <div class="bar-track" style="height:5px; background:rgba(0,0,0,0.5); border-radius:3px;">
                        <div class="bar-fill" style="width:${(alertness / 10) * 100}%; background:${alertColor}; box-shadow:0 0 8px ${alertColor}; transition: width 0.4s, background 0.4s; border-radius:3px;"></div>

            }
        } else if (phase === '终结回合') {

            borderColor = 'var(--ultimate-color)';
        } else if (phase === '战斗败北') {
            statusHtml = `<div style="display:flex; justify-content:space-between; color:var(--combat-color); font-weight:bold; font-size:0.8em; letter-spacing:1px; margin-bottom:4px;"><span><i class="fa-solid fa-heart-crack"></i> DEFEATED / 彻底力尽</span><span>Lv.${threat}</span></div>`;
            borderColor = 'var(--combat-color)';
        } else if (phase === '战斗胜利') {
            statusHtml = `<div style="display:flex; justify-content:space-between; color:var(--hp-color); font-weight:bold; font-size:0.8em; letter-spacing:1px; margin-bottom:4px;"><span><i class="fa-solid fa-trophy"></i> VICTORY / 战斗胜利</span><span>Lv.${threat}</span></div>`;
            borderColor = 'var(--hp-color)';
        } else {
            statusHtml = `<div style="display:flex; justify-content:space-between; color:var(--combat-color); font-weight:bold; font-size:0.8em; letter-spacing:1px; margin-bottom:4px;"><span><i class="fa-solid fa-swords"></i> COMBAT ACTIVE</span><span>Lv.${threat}</span></div>`;
            borderColor = 'var(--combat-color)';
        }

            const log = getValue(data, '主角.战斗临时.当前战斗日志', '无');
            if (log && log !== '无') {

                const res = (parts[0] || '未知').replace(/^(?:结果|战前|遭遇|抵抗|爆发|忍耐|迎合):/, '').trim();
                const det = parts.slice(1).filter(p => !/^\s*T:[A-Z_]+\s*$/.test(p)).join('<br>') || '';

                <div class="combat-log-btn" onclick="this.nextElementSibling.classList.toggle('show')">
                    <i class="fa-solid fa-terminal"></i> 检定终端: ${res} <i class="fa-solid fa-caret-down" style="margin-left:auto"></i>
                </div>
                <div class="combat-log-content">
                    <div>${res}</div>
                    <div style="opacity:0.7; margin-top:2px;">${det}</div>
                </div>`;
            }
        }

        return `
        <div class="boss-panel" style="border-left-color:${borderColor}">
            ${statusHtml}
            <div style="display:flex; justify-content:space-between; align-items:center;">
                <span style="font-weight:bold; font-size:1.05em; display:flex; align-items:center; gap:6px;">
                    ${name}
                    <span style="font-size:0.7em; font-weight:normal; padding:1px 6px; border-radius:4px; background:rgba(255,255,255,0.1); color:var(--text-dim);">
                        <i class="fa-solid ${typeIcon}"></i> ${cType}
                    </span>
                </span>
                <span style="font-weight:bold; color:${borderColor}; font-family:var(--font-mono);">${hp}%</span>
            </div>
            <div class="bar-track">
                <div class="phase-marker" style="left:30%"></div><div class="phase-marker" style="left:70%"></div>
                <div class="bar-fill" style="width:${hp}%; background:${borderColor};"></div>
            </div>
            <div style="font-size:0.85em; color:var(--text-dim); margin-top:2px;">
                <i class="fa-solid fa-tag"></i> ${themes.join(' / ') || '无数据'}
            </div>
            ${alertHtml}
            ${logHtml}
        </div>`;
    };

            } else if (isPurifyReady) {
                purifyBtnHtml = isFallen

                    : `<button class="purify-call-btn holy" onclick="triggerPurifyPregnancyEvent()"><i class="fa-solid fa-wand-magic-sparkles"></i> <span>✧ 圣洁涤荡 · 净化胚胎 (100%圣光) ✧</span></button>`;
            }

            const father = getValue(data, '主角.生理状态.子宫状态.孕育信息.孕育源', '未知魔物');
            
            let stageName = '', stageDesc = '', progressPct = Math.min(100, (count / 8) * 100);
            let stageColor = '#f472b6';

            if (count >= 7) {
                stageName = '🌸 生命成熟 ';
                stageColor = 'var(--ultimate-color)';
                stageDesc = isFallen
                    ? `腹内生命已彻底成熟，胎动异常活跃。这具极品肉体已随时准备好迎接新生命的诞生。`
                    : `小腹鼓胀至极，伴随着明显的阵痛与剧烈胎动，已处于临盆待产状态。`;
                
                const isBirthPending = getValue(data, '系统状态.待处理事件.分娩事件') === true;
                bottomWidgetHtml = isBirthPending
                    ? `<div style="text-align:center; margin-top:8px; font-size:0.8em; color:var(--ultimate-color); font-weight:bold;"><i class="fa-solid fa-spinner fa-spin"></i> 阵痛中... 发送消息即可迎来分娩</div>`
                    : `<button class="birth-call-btn" onclick="triggerBirthEvent()"><i class="fa-solid fa-baby"></i> <span>✦ 顺应分娩 · 诞下子嗣 ✦</span></button>`;
            } else if (count >= 4) {
                stageName = '🤰 显怀隆起';
                stageColor = '#f472b6';
                stageDesc = isFallen
                    ? `小腹呈现出引人注目的饱满隆起，能清晰感受到生命的欢快与共鸣。`
                    : `腰腹沉重，腹部已明显饱满隆起，走动时能察觉到胎动。`;
            } else {
                stageName = '🌱 生命萌芽';
                stageColor = '#a78bfa';
                stageDesc = isFallen
                    ? `子宫深处温热发烫，生命的种子已悄然扎根，正贪婪汲取魔力萌发。`

        } else if (postState === '恢复期') {

    // --- 轻量级非阻塞式浮动气泡 ---
    const showMaternalToast = (msg, icon = 'fa-wand-magic-sparkles', color = 'var(--ultimate-color)') => {
        try {
            const toastr = window.toastr || window.parent?.toastr || window.top?.toastr;
            if (toastr && typeof toastr.info === 'function') {
                toastr.info(msg, '孕育系统');
                return;
            }
        } catch(e){}

        let toastEl = document.getElementById('maternal-live-toast');
        if (!toastEl) {
            toastEl = document.createElement('div');
            toastEl.id = 'maternal-live-toast';
            toastEl.style.cssText = `
                position: fixed; bottom: 25px; left: 50%; transform: translateX(-50%) translateY(20px);

        toastEl.style.borderColor = color;
        toastEl.style.opacity = '1';

        toastEl._timer = setTimeout(() => {

            toastEl.style.transform = 'translateX(-50%) translateY(20px)';

    // --- MVU 变量更新（孕育卡片依赖） ---
    window.gm_update = async (path, value, mode = 'set') => {
        try {
            const Mvu = window.Mvu || window.parent?.Mvu || window.top?.Mvu;
            if (!Mvu) return;
            const mvuData = Mvu.getMvuData({ type: 'message', message_id: -1 });
            if (!mvuData) return;
            if (typeof _ !== 'undefined' && _.set) {
                if (mode === 'remove') _.unset(mvuData, path);
                else if (mode === 'add') {
                    const old = _.get(mvuData, path) || 0;
                    _.set(mvuData, path, Number(old) + Number(value));
                } else {
                    _.set(mvuData, path, value);
                }
            }
            await Mvu.replaceMvuData(mvuData, { type: 'message', message_id: -1 });
            const cleanData = JSON.parse(JSON.stringify(mvuData.stat_data || mvuData));
            document.dispatchEvent(new CustomEvent(Mvu.events.VARIABLE_UPDATE_ENDED, { detail: { stat_data: cleanData }, bubbles: true }));
        } catch (e) {
            console.error('[Mobile Status] gm_update error:', e);
        }
    };
