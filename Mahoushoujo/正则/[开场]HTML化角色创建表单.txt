                data['selected-kinks'] = Array.from(document.querySelectorAll('.kink-tag.selected')).map(el => el.textContent);
                return data;
            };

            const populateForm = (data) => {
                if (!data) return;
                allInputs.forEach(input => {
                    const key = (input.type === 'radio') ? input.name : (input.id || input.name);
                    if (data[key] !== undefined) {
                        if (input.type === 'radio') {
                           input.checked = (input.value === data[key]);
                        } else {
                           input.value = data[key];
                        }
                    } else {
                        if (input.type === 'text' || input.tagName === 'TEXTAREA') {
                            if (input.id === 'char-color-text') input.value = '樱粉色';
                            else if (input.id === 'char-ability') input.value = '操控光的能力';
                            else input.value = ''; 
                        } else if (input.type === 'number') {
                            if (input.id === 'start-month') input.value = '3';
                            else if (input.id === 'start-day') input.value = '5';
                            else if (input.id === 'start-fame') input.value = '0';
                            else if (input.id === 'start-rep') input.value = '0';
                        } else if (input.tagName === 'SELECT') {
                            input.selectedIndex = 0;
                        }
                    }
                });

                document.querySelectorAll('.kink-tag').forEach(el => el.classList.remove('selected'));
                if (data['selected-kinks'] && Array.isArray(data['selected-kinks'])) {
                    document.querySelectorAll('.kink-tag').forEach(el => {
                        if (data['selected-kinks'].includes(el.textContent)) el.classList.add('selected');
                    });
                }

                updateUIState();
                renderRepChips();
                try { colorPicker.value = colorText.value; } catch (err) {}
                setTimeout(() => {
                    textareas.forEach(ta => autoResizeTextarea(ta));
                }, 10);
            };

            const saveAutoSave = () => localStorage.setItem(AUTOSAVE_KEY, JSON.stringify(getCurrentFormData()));
            const debouncedAutoSave = () => {
                clearTimeout(saveTimeout);
                saveTimeout = setTimeout(saveAutoSave, 500);
            };
            
            const rescueOldData = () => {
                const oldAuto = localStorage.getItem('magicalGirl_AutoSave_v3');
                if (oldAuto && !localStorage.getItem(AUTOSAVE_KEY)) {
                    localStorage.setItem(AUTOSAVE_KEY, oldAuto);
                }
                const oldProfiles = localStorage.getItem('magicalGirl_ProfileManager_v3');
                if (oldProfiles && !localStorage.getItem(PROFILE_STORAGE_KEY)) {
                    localStorage.setItem(PROFILE_STORAGE_KEY, oldProfiles);
                }
            };

            const loadAutoSave = () => {
                const autoSavedData = localStorage.getItem(AUTOSAVE_KEY);
                if (autoSavedData) {
                    populateForm(JSON.parse(autoSavedData));
                } else {
                    updateUIState();
                    renderRepChips();
                }
            };
            
            const getProfiles = () => JSON.parse(localStorage.getItem(PROFILE_STORAGE_KEY)) || {};
            const saveProfiles = (profiles) => localStorage.setItem(PROFILE_STORAGE_KEY, JSON.stringify(profiles));

                exportSelectedProfileBtn.disabled = !profileSelect.value;

                profileSelect.innerHTML = '';
                if (profileNames.length === 0) {
                    profileSelect.innerHTML = '<option value="">-- 尚无已存方案 --</option>';
                    loadProfileBtn.disabled = true;
                    deleteProfileBtn.disabled = true;
                    exportProfilesBtn.disabled = true; 
                    exportSelectedProfileBtn.disabled = true;
                } else {
                    profileSelect.innerHTML = '<option value="">-- 选择方案 --</option>';
                    profileNames.sort().forEach(name => {
                        const option = document.createElement('option');
                        option.value = name;
                        option.textContent = name;
                        profileSelect.appendChild(option);
                    });
                    loadProfileBtn.disabled = false;
                    deleteProfileBtn.disabled = false;
                    exportProfilesBtn.disabled = false;
                    updateExportSelectedBtn();
                }

            colorPicker.addEventListener('input', () => { 
                colorText.value = colorPicker.value; 
                debouncedAutoSave();

                updateUIState();

                    populateDropdown();

                    showNotification(`保存成功: "${profileName.trim()}"`);

                    const profiles = getProfiles();

                    showNotification(`已删除: "${selectedName}"`);

                    return;

                    closeSignaturePopover();

                };

                if (Object.keys(profiles).length === 0) return;
                const a = document.createElement('a');
                a.href = URL.createObjectURL(new Blob([JSON.stringify(profiles, null, 2)], {type: 'application/json'}));
                a.download = `magical-girl-profiles-${Date.now()}.json`;
                a.click();
                URL.revokeObjectURL(a.href);
                showNotification('全部方案已导出');
            });

            importProfilesBtn.addEventListener('click', () => importFileInput.click());
            
            importFileInput.addEventListener('change', (event) => {
                const file = event.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onload = (e) => {
                    try {
                        const imported = JSON.parse(e.target.result);
                        const meta = imported[META_KEY];
                        const authorInfo = (meta && typeof meta === 'object' && meta.author) ? ` · 来自: ${meta.author}` : '';

                        let current = getProfiles(), count = 0;
                        for (const name in imported) {
                            if (name === META_KEY || name.startsWith('_')) continue;
                            const value = imported[name];
                            if (!value || typeof value !== 'object') continue;
                            if (!current[name] || confirm(`覆盖同名方案 "${name}" 吗？`)) { 
                                current[name] = value; 
                                count++; 
                            }
                        }
                        saveProfiles(current);
                        populateDropdown();
                        if (count > 0) {
                            showNotification(`导入了 ${count} 个方案${authorInfo}`);
                        } else {
                            showNotification('未找到有效方案', 'error');
                        }
                    } catch (error) { 
                        showNotification('导入失败：文件格式错误', 'error'); 
                    }
                    importFileInput.value = '';
                };
                reader.readAsText(file);
            });
            
            resetFormBtn.addEventListener('click', () => {
                if (confirm('确定要清空当前所有内容吗？')) {
                    form.reset();
                    document.getElementById('state-good').checked = true;
                    document.getElementById('time-random').checked = true;
                    document.getElementById('char-color-text').value = '樱粉色';
                    document.getElementById('char-color-picker').value = '#FFB6C1';
                    document.getElementById('char-ability').value = '操控光的能力';
                    document.getElementById('start-month').value = '3';
                    document.getElementById('start-day').value = '5';
                    document.getElementById('start-fame').value = '0';
                    document.getElementById('start-rep').value = '0';
                    document.querySelectorAll('.kink-tag').forEach(el => el.classList.remove('selected'));
                    updateUIState();
                    renderRepChips();
                    debouncedAutoSave();
                    showNotification('表单已重置');
                }
            });

            generateBtn.addEventListener('click', () => {
                saveAutoSave();
                generateBtn.disabled = true;
                generateBtn.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> <span>正在签订...</span>`;
                
                const data = getCurrentFormData();
                
                const buildPrunedYaml = (data) => {
                    const lines = [];
                    const add = (indent, text) => lines.push('  '.repeat(indent) + text);
                    const quote = (val) => `"${val.replace(/\r?\n/g, ' ').replace(/"/g, '\\"')}"`;
                    
                    const name = (data['char-name'] || '').trim();
                    const age = (data['char-age'] || '').trim();
                    const identity = (data['char-identity'] || '').trim();
                    const traits = (data['char-traits'] || '').trim();
                    const state = data['personality-state'] || '善良';
                    const master = (data['char-master'] || '').trim();
                    
const selectedKinks = (data['selected-kinks'] || []).filter(Boolean);
const kinks = selectedKinks.join('、');
const customKinks = (data['char-kinks'] || '').trim();

                    const mgName = (data['char-mg-name'] || '').trim();
                    const ability = (data['char-ability'] || '').trim();
                    const weapon = (data['char-weapon'] || '').trim();
                    const outfit = (data['char-outfit'] || '').trim();
                    const color = (data['char-color-text'] || '').trim();
                    const familiarType = (data['familiar-type'] || '').trim();
                    const familiarPers = (data['familiar-personality'] || '').trim();
                    const familiarRel = (data['familiar-relationship'] || '').trim();

                    const height = (data['char-height'] || '').trim();
                    const bodyCup = (data['char-body-cup'] || '').trim();
                    const sensitivity = (data['char-sensitivity'] || '').trim();
                    const appearance = (data['appearance-keywords'] || '').trim();
                    const bodyDetail = (data['body-keywords'] || '').trim();

                    const worldLore = (data['world-lore'] || '').trim();
                    const enemyLore = (data['enemy-lore'] || '').trim();
                    const extraNotes = (data['extra-notes'] || '').trim();

                    const timeMode = data['start-time-mode'] || '随机';
                    const startFame = Math.max(0, Math.min(100, parseInt(data['start-fame'], 10) || 0));
                    const startRep = Math.max(-100, Math.min(100, parseInt(data['start-rep'], 10) || 0));

                    add(0, "创角档案:");

                    add(1, "核心人格:");
                    add(2, `姓名: ${name ? quote(name) : '"[AI请填充]"'}`);
                    add(2, `人格路线: "${state}"`);
                    if (age) add(2, `年龄: ${quote(age)}`);
                    if (identity) add(2, `表世界身份: ${quote(identity)}`);
                    if (traits) add(2, `性格特质: ${quote(traits)}`);
                    if (state === '堕落' && master) add(2, `侍奉主人: ${quote(master)}`);
                    if (kinks) add(2, `初始性癖: ${quote(kinks)}`);
                    if (customKinks) add(2, `内容补充: ${quote(customKinks)}`);

                    add(1, "魔法契约:");
                    if (mgName) add(2, `魔法少女名: ${quote(mgName)}`);
                    if (color) add(2, `代表色与光效: ${quote(color)}`);
                    if (ability) add(2, `核心魔法能力: ${quote(ability)}`);
                    if (weapon) add(2, `专属武装: ${quote(weapon)}`);
                    if (outfit) add(2, `变身服材质风格: ${quote(outfit)}`);

                    if (familiarType || familiarPers || familiarRel) {
                        add(2, "使魔伙伴:");
                        if (familiarType) add(3, `种类形态: ${quote(familiarType)}`);
                        if (familiarPers) add(3, `性格表现: ${quote(familiarPers)}`);
                        if (familiarRel) add(3, `羁绊关系: ${quote(familiarRel)}`);

                        const mm = Math.max(1, Math.min(12, parseInt(data['start-month'], 10) || 3));
                        const dd = Math.max(1, Math.min(31, parseInt(data['start-day'], 10) || 5));
                        const mmdd = mm * 100 + dd;
                        add(2, `设定模式: "手动指定"`);

                        add(2, `初始化指令: "请在变量初始化时，将 /世界/日期 设为 ${mmdd}。此项优先级高于世界书提供的随机日期默认值。"`);

                        add(2, `设定模式: "随机"`);
                        add(2, `初始化指令: "请使用世界书提供的随机日期机制，为 /世界/日期 生成随机初始值。"`);

                    add(2, `知名度: ${startFame}`);

                        add(1, "世界设定与环境:");
                        
                        const addMultilineBlock = (label, text) => {
                            const linesArr = text.split(/\r?\n/).map(l => l.trim()).filter(Boolean);
                            if (linesArr.length === 1) {
                                add(2, `${label}: ${quote(linesArr[0])}`);
                            } else if (linesArr.length > 1) {
                                add(2, `${label}: |`);
                                linesArr.forEach(l => add(3, l));
                            }
                        };

                        if (worldLore) addMultilineBlock("世界观背景", worldLore);
                        if (enemyLore) addMultilineBlock("敌对势力与机制", enemyLore);
                        if (extraNotes) addMultilineBlock("补充规则、设定与约束", extraNotes);
                    }

                    return lines.join('\n');
                };

                const yamlString = `<init_data>\n\`\`\`yaml\n# [创角初始化指令]\n# 1. 凡未标注 [AI请填充] 的字段均为用户确认的设定，必须严格遵循并写入变量。\n# 2. 未列出的属性或标注 [AI请填充] 的字段，请基于美学进行富有张力的合理推演。\n# 3. 【开局时间】若为手动指定，必须在初始化时将 /世界/日期 设为指定值，优先级高于世界书随机日期。\n# 4. 【开局声望】请按 [生成] 世界书中的声望初始化协议，派生等级并写入变量。\n\n${buildPrunedYaml(data)}\n\`\`\`\n</init_data>`;
                
                setTimeout(() => {
                    try {
                        const inputArea = window.parent.document.querySelector('#send_textarea');
                        const sendButton = window.parent.document.querySelector('#send_but');
                        if (inputArea && sendButton) {
                            inputArea.value = yamlString;
                            inputArea.dispatchEvent(new Event('input', { bubbles: true }));
                            sendButton.click();
                            showNotification('契约已发送！等待AI回应...');
                            generateBtn.innerHTML = `<i class="fa-solid fa-check"></i> <span>契约已发送</span>`;
                        } else { 
                            throw new Error('SillyTavern UI not found.'); 
                        }
                    } catch (error) {
                        console.error('Error sending message:', error);
                        generateBtn.innerHTML = `<i class="fa-solid fa-xmark"></i> <span>发送失败</span>`;
                        navigator.clipboard.writeText(yamlString).then(() => {
                            showNotification('自动发送失败！已复制到剪贴板', 'error');
                            alert('自动发送失败！角色信息已复制到您的剪贴板，请手动粘贴并发送。');
                        });
                    }
                    setTimeout(() => {
                        generateBtn.disabled = false;
                        generateBtn.innerHTML = initialButtonHTML;
                    }, 4000);
                }, 400);
            });

            rescueOldData();
            populateDropdown(); 
            loadAutoSave(); 
            loadSignature();
        });
