        const resolveActiveConfig = (stat) => {
            const phase = getDeep(stat, '进程.阶段', '幕间休息');
            const hEventName = getDeep(stat, '世界.幕间H事件.名称', '无');
            if (phase === '战斗回合' || phase === '终结回合') return BUTTON_CONFIGS['战斗回合'];
            if (phase === '战斗准备') return BUTTON_CONFIGS['战斗准备'];
            if (phase === '幕间休息' && hEventName !== '无') return BUTTON_CONFIGS['H事件'];
            return null;
        };

        // ── 更新战术变量 ──
        const setTactic = async (tactic) => {
            try {
                const Mvu = getMvu();
                if (!Mvu) return;
                const mvuData = Mvu.getMvuData({ type: 'message', message_id: -1 });
                if (!mvuData) return;

                if (typeof _ !== 'undefined' && _.set) {
                    _.set(mvuData, 'stat_data.主角.战斗临时.战斗方式', tactic);
                } else {
                    mvuData.stat_data.主角.战斗临时.战斗方式 = tactic;
                }

                await Mvu.replaceMvuData(mvuData, { type: 'message', message_id: -1 });
                const cleanData = JSON.parse(JSON.stringify(mvuData.stat_data || mvuData));
                document.dispatchEvent(new CustomEvent(Mvu.events.VARIABLE_UPDATE_ENDED, {
                    detail: { stat_data: cleanData },
                    bubbles: true
                }));

                document.querySelectorAll('#tactics-section .tactic-btn').forEach(btn => {
                    btn.classList.toggle('active', btn.dataset.tactic === tactic);
                });
            } catch (e) {
                console.error('[QuickOptions] setTactic error:', e);
            }
        };

        // ── 渲染战术区 ──
        // ══════════════════════════════════════════
        // ██ 隐藏翻转：背面「撤退 / 投降」（仅战斗回合可用）
        // ══════════════════════════════════════════
        const TC_FLIP_MS = 190;
        let _tcFlipped = false;

        // 读取当前锁定的战术（从【正面】按钮的 active 态反读）
        // 注意：工单原稿的 section.querySelector('.tc-back-btn.active').dataset.tactic
        // 取不到值 —— data-tactic 只挂在正面的 .tactic-btn 上，背面 .tc-back-btn 没有。
        // 因此这里必须查 .tactic-btn。
        const getCurrentTactic = () => {
            const el = document.querySelector('.tactic-btn.active');
            return el && el.dataset ? el.dataset.tactic : '';
        };

        const syncSecretState = (currentTactic) => {
            const section = document.getElementById('tactics-section');
            if (!section) return;
            const flee = section.querySelector('.tc-back-btn.flee');
            const giveup = section.querySelector('.tc-back-btn.giveup');
            if (flee) flee.classList.toggle('active', currentTactic === '撤退');
            if (giveup) giveup.classList.toggle('active', currentTactic === '投降');
        };

        const flipSecret = (toBack) => {
            const section = document.getElementById('tactics-section');
            if (!section || _tcFlipped === toBack) return;
            section.style.transform = 'rotateY(88deg)';
            section.style.opacity = '0';
            setTimeout(() => {
                _tcFlipped = toBack;
                section.classList.toggle('is-flipped', toBack);
                section.style.transform = 'rotateY(-88deg)';
                requestAnimationFrame(() => requestAnimationFrame(() => {
                    section.style.transform = '';
                    section.style.opacity = '';
                }));
            }, TC_FLIP_MS);
        };

        const ensureSecretUI = (section) => {
            if (!section.querySelector('.tc-backface')) {
                const back = document.createElement('div');
                back.className = 'tc-backface';

                const flee = document.createElement('button');
                flee.className = 'tc-back-btn flee';
                flee.title = '中断交战、拉开距离脱离战场。判定失败会被追击反打，而且越逃越难。';
                flee.innerHTML = '<i class="fa-solid fa-person-running"></i><span>撤退</span>';
                flee.addEventListener('click', () => { setTactic('撤退'); syncSecretState('撤退'); });

                const giveup = document.createElement('button');
                giveup.className = 'tc-back-btn giveup';
                giveup.title = '主动放弃战斗，直接判定战败并被俘 —— 不可撤销。';
                giveup.innerHTML = '<i class="fa-solid fa-flag"></i><span>投降</span>';
                giveup.addEventListener('click', () => {
                    if (!confirm('确定投降？\n\n将直接判定战败并被俘，此操作不可撤销。')) return;
                    setTactic('投降');
                    syncSecretState('投降');
                });

                // ── 返回按钮 [MG_FLIP_V2] ──
                // 原实现把返回键类名设成与正面小圆点同一族的「隐形密钮」类，
                // 而 CSS 里那个类带 display:none；JS 只给正面的小圆点补过
                // inline display:flex，从没给返回键补过 —— 于是「返回」按钮被
                // CSS 永久隐藏，玩家一旦翻到背面就再也回不到正面（死按钮）。
                // 现改为正规实体按钮：走 .tc-back-btn 的 flex 布局 + 文字标签。
                const ret = document.createElement('button');
                ret.type = 'button';
                ret.className = 'tc-back-btn back-return';
                ret.title = '取消当前选择并返回常规战术';
                ret.innerHTML = '<i class="fa-solid fa-arrow-left"></i><span>返回</span>';
                ret.addEventListener('click', () => {
                    // 只在「会让 render 再次自动翻回背面」的选择上重置，解除死锁：
                    //   撤退 / 投降 → 任何 render 都会强制翻回背面
                    //   空值        → 没有合法选择，回到阵营默认战术
                    // 若当前是 主动/牵制/推进/危机 等正面战术，则只关闭背面、不动选择。
                    const cur = getCurrentTactic();
                    if (cur === '撤退' || cur === '投降' || !cur) {
                        if (typeof setTactic === 'function') {
                            setTactic('主动');
                        } else if (typeof window.setCombatTactic === 'function') {
                            window.setCombatTactic('主动');
                        }
                    }
                    flipSecret(false);
                });

                back.appendChild(flee);
                back.appendChild(giveup);
                back.appendChild(ret);
                section.appendChild(back);
            }

            if (!section.querySelector('#tc-secret-dot')) {
                // ── 正面翻转开关 [MG_FLIP_V2] ──
                // 原实现继承 .tc-secret 的 display:none，仅靠 render 补一个
                // inline display:flex 才出现；图标 opacity 0.12 / font-size 0.5em，
                // 实际是肉眼不可见的灰尘点。现改为有边框的半透明功能键。
                // id 仍沿用 tc-secret-dot，以免影响既有显示/隐藏逻辑。
                const dot = document.createElement('div');
                dot.id = 'tc-secret-dot';
                dot.className = 'tc-toggle-btn';
                dot.title = '展开「撤退 / 投降」选项';
                dot.innerHTML = '<i class="fa-solid fa-angles-right"></i>';
                dot.addEventListener('click', () => flipSecret(true));
                section.appendChild(dot);
            }
        };

        const renderTactics = (stat) => {
            const section = document.getElementById('tactics-section');
            if (!section) return;

            const config = resolveActiveConfig(stat);
            if (!config) {
                section.classList.remove('visible', 'is-flipped');
                section.innerHTML = '';
                section.dataset.sig = '';
                _tcFlipped = false;
                return;
            }

            const isFallen = getDeep(stat, '主角.核心状态.人格状态', '善良') === '堕落';
            const buttons = config.getButtons ? config.getButtons(isFallen) : (config.buttons || []);
            const currentTactic = getDeep(stat, '主角.战斗临时.战斗方式', buttons[0]?.key || '');
            const ultPower = parseInt(
                getDeep(stat, isFallen ? '主角.核心状态.污秽魔力' : '主角.核心状态.净化之力', 0),
                10
            );
            const isUltReady = ultPower >= 100;

            // 只在按钮结构变化时重建 DOM，避免闪烁
            const signature = buttons.map(b => b.key).join('|') + '_' + isFallen + '_' + isUltReady;
            if (section.dataset.sig !== signature) {
                section.dataset.sig = signature;
                section.innerHTML = `<div class="section-label"><i class="fa-solid ${config.icon}"></i> 战术选择</div>`;

                buttons.forEach(b => {
                    const btn = document.createElement('button');
                    btn.className = 'tactic-btn ' + b.cls;
                    btn.dataset.tactic = b.key;
                    btn.title = b.tip;

                    const iconEl = document.createElement('i');
                    iconEl.className = 'fa-solid ' + b.icon;
                    btn.appendChild(iconEl);

                    const labelEl = document.createElement('span');
                    labelEl.textContent = b.short;
                    btn.appendChild(labelEl);

                    if (b.isUlt && !isUltReady) btn.disabled = true;

                    btn.addEventListener('click', () => setTactic(b.key));
                    section.appendChild(btn);
                });
            }

            // 每次渲染同步 active 状态
            section.querySelectorAll('.tactic-btn').forEach(btn => {
                btn.classList.toggle('active', btn.dataset.tactic === currentTactic);
            });

            // ── 隐藏翻转装配（仅战斗回合）──
            const isCombatRound = getDeep(stat, '进程.阶段', '') === '战斗回合';
            ensureSecretUI(section);
            const _dot = document.getElementById('tc-secret-dot');
            if (_dot) _dot.style.display = isCombatRound ? 'flex' : 'none';
            if (!isCombatRound && _tcFlipped) flipSecret(false);
            // ── 死锁解除 [MG_FLIP_V2] ──
            // 原实现：只要 currentTactic 是 撤退/投降 就强制翻回背面。
            // 玩家翻回正面后，下一次 render（变量同步 / 阶梯轮询）会立刻
            // 再次 flipSecret(true) —— 形成「翻不回正面」的死锁。
            // 现改为「跟随选择」而非「强制锁死」：
            //   · 选中 撤退/投降 → 翻到背面，让玩家看见自己的选择
            //   · 改选其它战术   → 自动翻回正面
            //   · 点「返回」重置为 主动 → 保持正面，不再被拉回去
            if (isCombatRound) {
                const _cur = getCurrentTactic();
                const _wantsBack = (_cur === '撤退' || _cur === '投降');
                if (_wantsBack !== _tcFlipped) flipSecret(_wantsBack);
            } else if (_tcFlipped) {
                flipSecret(false);
            }
            syncSecretState(currentTactic);

            section.classList.add('visible');
        };

        // ── 渲染 AI 建议选项 ──
        const OPTION_STYLES = [
            { className: 'style-radical', icon: 'fa-solid fa-feather-pointed' },
            { className: 'style-calm',    icon: 'fa-regular fa-snowflake' },
            { className: 'style-magic',   icon: 'fa-solid fa-wand-magic-sparkles' },
            { className: 'style-desire',  icon: 'fa-regular fa-heart' }
        ];

        const handleSend = (text) => {
            try {
                const inputArea = window.parent.document.querySelector('#send_textarea');
                if (inputArea) {
                    inputArea.value = (inputArea.value.trim() ? inputArea.value + '\\n' : '') + text;
                    inputArea.dispatchEvent(new Event('input', { bubbles: true }));
                }
            } catch (error) {
                console.error(error);
            }
        };

        const renderOptions = () => {
            const rawContainer = document.getElementById('options-container');
            if (!rawContainer) return;
            const rawText = rawContainer.textContent;
            rawContainer.innerHTML = '';

            const options = (rawText || '').split(/\r?\n/).map(s => s.trim()).filter(s => s.length > 0);
            if (options.length === 0) return;

            rawContainer.style.display = 'flex';

            options.forEach((text, index) => {
                const config = OPTION_STYLES[index % OPTION_STYLES.length];

                const button = document.createElement('button');
                button.className = 'option-btn ' + config.className;

                const iconBox = document.createElement('div');
                iconBox.className = 'icon-box';
                const icon = document.createElement('i');
                icon.className = config.icon;
                iconBox.appendChild(icon);

                const textSpan = document.createElement('span');
                textSpan.className = 'option-text';
                textSpan.textContent = text;

                button.appendChild(iconBox);
                button.appendChild(textSpan);
                button.onclick = () => handleSend(text);
                rawContainer.appendChild(button);
            });
        };

        // ── MVU 监听（延迟重试直到 MVU 就绪） ──
        const initMvuListener = () => {
            const Mvu = getMvu();
            if (!Mvu || !Mvu.events) {
                setTimeout(initMvuListener, 400);
                return;
            }

            document.addEventListener(Mvu.events.VARIABLE_UPDATE_ENDED, (e) => {
                const stat = e.detail?.stat_data;
                if (stat) renderTactics(stat);
            });

            try {
                const wrapper = Mvu.getMvuData({ type: 'message', message_id: -1 });
                const stat = wrapper?.stat_data || wrapper;
                if (stat) renderTactics(stat);
            } catch (e) {}
        };

        // ── 启动 ──
        document.addEventListener('DOMContentLoaded', () => {
            renderOptions();
            initMvuListener();
        });
    </script>
</body>
</html>
