        (() => {
            const getDeep = (obj, path, def = null) =>
                path.split('.').reduce((acc, part) => acc && acc[part], obj) ?? def;

            const getConsoleAncestors = () => {
                const list = [];
                const push = (d) => { if (d && list.indexOf(d) === -1) list.push(d); };
                try { push(window.parent && window.parent.document); } catch (e) {}
                try { push(window.top && window.top.document); } catch (e) {}
                return list;
            };

            const getMsgId = () => {
                try {
                    if (typeof getCurrentMessageId === 'function') {
                        const id = Number(getCurrentMessageId());
                        if (Number.isFinite(id)) return id;
                    }
                } catch (e) {}
                try {
                    const m = /^TH-message--(\d+)$/.exec((window.frameElement && window.frameElement.id) || '');
                    if (m) return Number(m[1]);
                } catch (e) {}
                return null;
            };
            const MSG_ID = getMsgId();

            const findMyMessage = () => {
                const frame = window.frameElement;
                if (!frame) return null;
                for (const doc of getConsoleAncestors()) {
                    if (MSG_ID !== null) {
                        const exact = doc.querySelector('#chat > .mes[mesid="' + MSG_ID + '"]');
                        if (exact) return exact;
                    }
                    let node = frame.parentElement;
                    while (node && node !== doc.documentElement) {
                        if (node.classList && (node.classList.contains('mes_text') || node.classList.contains('mes'))) return node;
                        node = node.parentElement;
                    }
                    const legacy = frame.closest && frame.closest('.mes_text, .mes, .chat-message');
                    if (legacy) return legacy;
                }
                return null;
            };

            const updateHostHeight = () => {
                try {
                    const frame = window.frameElement;
                    const panel = document.getElementById('tactics-container');
                    if (!frame || !panel || panel.style.display === 'none') return;
                    const apply = () => {
                        try {
                            const h = Math.max(panel.scrollHeight || 0, panel.offsetHeight || 0, 36);
                            frame.style.setProperty('height', (h + 4) + 'px', 'important');
                        } catch (e) {}
                    };
                    if (typeof requestAnimationFrame === 'function') {
                        requestAnimationFrame(() => requestAnimationFrame(apply));
                    } else {
                        setTimeout(apply, 32);
                    }
                } catch (e) {}
            };

            const toggleHostCollapse = (show) => {
                try {
                    const frame = window.frameElement;
                    if (!frame) return;
                    if (show) {
                        frame.style.removeProperty('display');
                        frame.style.removeProperty('margin');
                        frame.style.removeProperty('padding');
                        updateHostHeight();
                    } else {
                        frame.style.setProperty('display', 'none', 'important');
                        frame.style.setProperty('height', '0px', 'important');
                        frame.style.setProperty('margin', '0px', 'important');
                        frame.style.setProperty('padding', '0px', 'important');
                    }
                } catch(e) {}
            };

            try {
                if (typeof ResizeObserver === 'function') {
                    const _tcRO = new ResizeObserver(() => updateHostHeight());
                    const _tcWrap = document.getElementById('tactics-container');
                    if (_tcWrap) _tcRO.observe(_tcWrap);
                }
            } catch (e) {}

            window.setCombatTactic = async (tactic) => {
                try {

                    try { pushT(window.top && window.top.document); } catch (e) {}

                    for (const scope of containers) {
                        let frames = [];
                        try { frames = scope.querySelectorAll('iframe'); } catch (e) { frames = []; }
                        for (const f of frames) {
                            if (f === frame) continue;

                                const d = f.contentDocument;
                                if (!d) continue;
                                if (!d.getElementById('options-container')) continue;

                    const curTactic = getDeep(window._lastStat, '主角.战斗临时.战斗方式', '');
                    if (curTactic === '撤退' || curTactic === '投降') {
                        window.setCombatTactic('主动');
                    }
                    switchDeck(false);
                };

                if (cardFlee) cardFlee.onclick = () => {
                    window.setCombatTactic('撤退');
                };
                if (cardGiveup) cardGiveup.onclick = () => {
                    window.setCombatTactic('投降');
                };
            };

            let _tcSig = null;
            const render = (stat) => {
                const container = document.getElementById('tactics-container');
                if (!container || !stat || typeof stat !== 'object') return;
                window._lastStat = stat;

                if (hasQuickOptionsSibling()) {
                    if (container.style.display !== 'none') {
                        _tcSig = null;
                        container.style.display = 'none';
                        toggleHostCollapse(false);
                    }
                    return;
                }

                const config = getActiveConfig(stat);
                if (!config) {
                    container.style.display = 'none';
                    toggleHostCollapse(false);
                    return;
                }

                container.style.display = 'block';
                toggleHostCollapse(true);

                const phase = getDeep(stat, '进程.阶段', '');
                const isCombatRound = (phase === '战斗回合');
                const isFallen = getDeep(stat, '主角.核心状态.人格状态', '善良') === '堕落';
                const buttons = config.getButtons ? config.getButtons(isFallen) : [];
                const currentTactic = getDeep(stat, '主角.战斗临时.战斗方式', buttons[0]?.key || '');
                const fleeAttempts = parseInt(getDeep(stat, '主角.战斗临时.撤退尝试', 0), 10);

                // 徽章
                const badge = document.getElementById('tc-badge');
                if (badge) {
                    badge.innerHTML = `<i class="fa-solid ${config.icon}"></i>`;
                    badge.style.color = config.color;
                }

                // 只有战斗回合才展示抉择入口
                const btnOpen = document.getElementById('btn-open-decision');
                if (btnOpen) {
                    btnOpen.style.display = isCombatRound ? 'flex' : 'none';
                }

                // 动态构建常规战术按键
                const grid = document.getElementById('tactics-grid');
                const signature = buttons.map(b => b.key).join('|');
                if (grid.dataset.signature !== signature) {
                    grid.dataset.signature = signature;
                    grid.innerHTML = '';
                    buttons.forEach(b => {
                        const btn = document.createElement('button');
                        btn.className = 'tactic-btn ' + b.cls;
                        btn.dataset.tactic = b.key;
                        btn.title = b.tip;
                        btn.innerHTML = `<i class="fa-solid ${b.icon}"></i><span>${b.short}</span>`;
                        btn.onclick = () => window.setCombatTactic(b.key);
                        grid.appendChild(btn);
                    });
                }

                // 同步常规按键高亮
                const ultPower = parseInt(getDeep(stat, isFallen ? '主角.核心状态.污秽魔力' : '主角.核心状态.净化之力', 0), 10);
                const isUltReady = ultPower >= 100;
                grid.querySelectorAll('.tactic-btn').forEach(btn => {
                    const key = btn.dataset.tactic;
                    const cfg = buttons.find(b => b.key === key);
                    btn.classList.toggle('active', key === currentTactic);
                    if (cfg && cfg.isUlt) {
                        btn.disabled = !isUltReady;
                        btn.classList.toggle('ready', isUltReady);
                    } else {
                        btn.disabled = false;
                    }
                });

                // 同步战局决断卡片状态
                const cardFlee = document.getElementById('card-flee');
                const cardGiveup = document.getElementById('card-giveup');
                const fleeTag = document.getElementById('flee-tag');

                if (cardFlee) cardFlee.classList.toggle('active', currentTactic === '撤退');
                if (cardGiveup) cardGiveup.classList.toggle('active', currentTactic === '投降');
                
                // 动态显示撤退惩罚点数
                if (fleeTag) {
                    if (fleeAttempts > 0) {
                        fleeTag.innerText = `惩罚 -${fleeAttempts * 3}`;
                        fleeTag.style.background = 'rgba(248, 113, 113, 0.2)';
                        fleeTag.style.color = '#ff7675';
                    } else {
                        fleeTag.innerText = '机动检定';
                        fleeTag.style.background = 'rgba(224, 175, 104, 0.15)';
                        fleeTag.style.color = 'var(--tc-flee)';
                    }
                }

                // 若当前状态是撤退或投降，直接展示决断页；否则展示战斗页
                const wantsDecisionDeck = isCombatRound && (currentTactic === '撤退' || currentTactic === '投降');
                if (wantsDecisionDeck !== _isDecisionMode) {
                    switchDeck(wantsDecisionDeck);
                }
            };

            const getMvu = () => (window.Mvu || window.parent?.Mvu || window.top?.Mvu || null);

            const pullStatData = () => {
                const Mvu = getMvu();
                if (!Mvu || typeof Mvu.getMvuData !== 'function') return null;
                for (const o of [{ type: 'message', message_id: -1 }, { type: 'message', message_id: 'latest' }]) {
                    try {
                        const wrapper = Mvu.getMvuData(o);
                        const stat = wrapper?.stat_data || wrapper;
                        if (stat && typeof stat === 'object') return stat;
                    } catch (e) {}
                }
                return null;
            };

            const fetchAndRender = () => {
                const stat = pullStatData();
                if (stat) render(stat);
                updateHostHeight();
            };

            const init = () => {
                initDecisionCards();
                const Mvu = getMvu();
                if (Mvu && Mvu.events) {
                    try {
                        document.addEventListener(Mvu.events.VARIABLE_UPDATE_ENDED, (e) => {
                            const stat = e?.detail?.stat_data;
                            if (stat) { _tcSig = null; render(stat); updateHostHeight(); }
                        });
                    } catch (e) {}
                } else {
                    setTimeout(init, 300);
                }

                fetchAndRender();
                [100, 300, 800, 1500].forEach((delay) => setTimeout(fetchAndRender, delay));

                try {
                    if (document.readyState === 'loading') {
                        document.addEventListener('DOMContentLoaded', fetchAndRender);
                    }
                    window.addEventListener('load', fetchAndRender);
                    window.addEventListener('resize', () => updateHostHeight());
                } catch (e) {}
            };

            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', init);
            } else {
                init();
            }
        })();
    </script>
</body>
</html>
