<body>
    <div class="options-container" id="options-container" style="display:none;">$1</div>

    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const rawContainer = document.getElementById('options-container');
            if (!rawContainer) return;
            
            const rawText = rawContainer.textContent;
            rawContainer.innerHTML = '';
            rawContainer.style.display = 'flex';

            const optionStyles = [
                { className: 'style-0', icon: 'fa-solid fa-heart' },
                { className: 'style-1', icon: 'fa-solid fa-cloud' },
                { className: 'style-2', icon: 'fa-solid fa-star' },
                { className: 'style-3', icon: 'fa-solid fa-leaf' }
            ];

            const options = rawText.split(/\r?\n/).map(s => s.trim()).filter(s => s.length > 0);

            const handleSend = (text) => {
                try {
                    const inputArea = window.parent.document.querySelector('#send_textarea');
                    if (inputArea) {
                        inputArea.value = (inputArea.value.trim() ? inputArea.value + '\\n' : '') + text;
                        inputArea.dispatchEvent(new Event('input', { bubbles: true }));
                    }
                } catch (error) { console.error(error); }
            };
            
            options.forEach((text, index) => {
                const config = optionStyles[index % optionStyles.length]; 

                const button = document.createElement('button');
                button.className = 'option-btn ' + config.className;
                
                const iconBox = document.createElement('div');
                iconBox.className = 'icon-box';
                const icon = document.createElement('i');
                icon.className = config.icon;
                iconBox.appendChild(icon);

                const textSpan = document.createElement('span');
                textSpan.className = 'option-text';
                textSpan.textContent = text;
                
                button.appendChild(iconBox);
                button.appendChild(textSpan);
                
                button.onclick = () => handleSend(text);
                rawContainer.appendChild(button);
            });
        });
    </script>
</body>
