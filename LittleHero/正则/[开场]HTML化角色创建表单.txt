        document.addEventListener('DOMContentLoaded', () => {
            const form = document.getElementById('creator-form');
            const generateBtn = document.getElementById('generate-btn');
            const resetFormBtn = document.getElementById('reset-form-btn');
            const notification = document.getElementById('copy-notification');
            const creatorCard = document.getElementById('creator-card');
            const colorPicker = document.getElementById('char-color-picker');
            const colorText = document.getElementById('char-color-text');
            const personalityRadios = document.querySelectorAll('input[name="personality-state"]');
            const masterField = document.getElementById('master-field');
            const initialButtonHTML = generateBtn.innerHTML;
            const allInputs = Array.from(form.querySelectorAll('input, textarea, select'));

            const profileSelect = document.getElementById('profile-select');
            const loadProfileBtn = document.getElementById('load-profile-btn');
            const saveProfileBtn = document.getElementById('save-profile-btn');
            const deleteProfileBtn = document.getElementById('delete-profile-btn');
            const exportProfilesBtn = document.getElementById('export-profiles-btn');
            const importProfilesBtn = document.getElementById('import-profiles-btn');
            const importFileInput = document.getElementById('import-file-input');

            // --- 配置 ---
            const AUTOSAVE_KEY = 'magicalGirl_AutoSave_v2';
            const PROFILE_STORAGE_KEY = 'magicalGirl_ProfileManager_v2';

            // 兼容性 Key (用于数据搜救)
            const OLD_KEYS = [
                'magicalGirl_AutoSave',             // v1.0
                'magicalGirl_ProfileManager',       // v1.0
                'magicalGirl_AutoSave_v4',          // v2.4.2 临时版
                'magicalGirl_Profiles_v4'           // v2.4.2 临时版
            ];

            let saveTimeout;

            // --- 文本框高度自适应逻辑 ---
            const textareas = document.querySelectorAll('textarea');
            const autoResizeTextarea = (ta) => {
                ta.style.height = 'auto'; // 先归零
                ta.style.height = ta.scrollHeight + 'px'; // 根据内容撑开
            };
            textareas.forEach(ta => {
                ta.addEventListener('input', () => autoResizeTextarea(ta));
            });

            const showNotification = (message, color = 'var(--success-color)') => {
                notification.textContent = message;
                notification.style.color = color;
                notification.style.opacity = '1';
                clearTimeout(notification.timeoutId);
                notification.timeoutId = setTimeout(() => { notification.style.opacity = '0'; }, 3500);
            };

            const updateUIState = () => {
                const selectedRadio = document.querySelector('input[name="personality-state"]:checked');
                const isFallen = selectedRadio?.value === '反派';
                creatorCard.classList.toggle('theme-fallen', isFallen);
                if (isFallen) {
                    masterField.classList.add('visible');
                } else {
                    masterField.classList.remove('visible');
                }
            };

            const getCurrentFormData = () => {
                const data = {};
                allInputs.forEach(input => {
                    if (input.type === 'radio') {
                        if (input.checked) data[input.name] = input.value;
                    } else if (input.id) {
                        data[input.id] = input.value;
                    }
                });
                return data;
            };

// --- [核心修复] 兼容性数据填充 ---
            const populateForm = (data) => {
                if (!data) return;
                allInputs.forEach(input => {
                    // 逻辑: 如果是Radio，使用NAME查找；如果是Input，优先用ID。
                    const key = (input.type === 'radio') ? input.name : (input.id || input.name);

                    if (data[key] !== undefined) {
                        // 情况A: 存档里有这个数据 -> 正常读取
                        if (input.type === 'radio') {
                           input.checked = (input.value === data[key]);
                        } else {
                           input.value = data[key];
                        }
                    } else {
                        // 情况B: 存档里没这个数据 (旧存档) -> 强制重置为默认值
                        // 防止旧档读取后，新字段(如性癖)残留着上一次填写的内容
                        if (input.type === 'text' || input.tagName === 'TEXTAREA') {
                            // 特殊默认值处理
                            if (input.id === 'char-color-text') input.value = '樱粉色';
                            else if (input.id === 'char-ability') input.value = '操控光的能力';
                            else input.value = ''; // 其他所有文本框清空
                        }
                        else if (input.type === 'color') {
                            input.value = '#FFB6C1';
                        }
                        // Radio 按钮保持原状或默认第一个，通常不需要在这里强制重置
                    }
                });

                // 强制更新UI视觉
                updateUIState();
                try { colorPicker.value = colorText.value; } catch (err) {}
                setTimeout(() => { textareas.forEach(ta => autoResizeTextarea(ta)); }, 10);
            };

            const saveAutoSave = () => localStorage.setItem(AUTOSAVE_KEY, JSON.stringify(getCurrentFormData()));
            const debouncedAutoSave = () => { clearTimeout(saveTimeout); saveTimeout = setTimeout(saveAutoSave, 500); };

            const loadAutoSave = () => {
                const autoSavedData = localStorage.getItem(AUTOSAVE_KEY);
                if (autoSavedData) {
                    populateForm(JSON.parse(autoSavedData));
                } else {
                    updateUIState();
                }
            };

            // --- 数据搜救 ---
            const rescueOldData = () => {
                let rescuedCount = 0;
                const currentProfiles = JSON.parse(localStorage.getItem(PROFILE_STORAGE_KEY) || '{}');
                const v1Profiles = localStorage.getItem('magicalGirl_ProfileManager');
                const v4Profiles = localStorage.getItem('magicalGirl_Profiles_v4');

                if (v1Profiles || v4Profiles) {
                    const p1 = v1Profiles ? JSON.parse(v1Profiles) : {};
                    const p4 = v4Profiles ? JSON.parse(v4Profiles) : {};
                    const merged = { ...p1, ...p4, ...currentProfiles };
                    if (Object.keys(merged).length > Object.keys(currentProfiles).length) {
                        localStorage.setItem(PROFILE_STORAGE_KEY, JSON.stringify(merged));
                        rescuedCount += (Object.keys(merged).length - Object.keys(currentProfiles).length);
                    }
                }
                if (!localStorage.getItem(AUTOSAVE_KEY)) {
                    const v4Auto = localStorage.getItem('magicalGirl_AutoSave_v4');
                    const v1Auto = localStorage.getItem('magicalGirl_AutoSave');
                    if (v4Auto) { localStorage.setItem(AUTOSAVE_KEY, v4Auto); populateForm(JSON.parse(v4Auto)); rescuedCount++; }
                    else if (v1Auto) { localStorage.setItem(AUTOSAVE_KEY, v1Auto); populateForm(JSON.parse(v1Auto)); rescuedCount++; }
                }
                if (rescuedCount > 0) showNotification(`已成功找回并迁移了旧版数据！`, 'var(--success-color)');
            };

            const getProfiles = () => JSON.parse(localStorage.getItem(PROFILE_STORAGE_KEY)) || {};
            const saveProfiles = (profiles) => localStorage.setItem(PROFILE_STORAGE_KEY, JSON.stringify(profiles));

            const populateDropdown = () => {
                const profiles = getProfiles();
                const profileNames = Object.keys(profiles);

                profileSelect.innerHTML = '';
                if (profileNames.length === 0) {
                    profileSelect.innerHTML = '<option value="">-- 尚无已存方案 --</option>';
                    loadProfileBtn.disabled = true;
                    deleteProfileBtn.disabled = true;
                    exportProfilesBtn.disabled = true;
                    importProfilesBtn.disabled = false;
                } else {
                    profileSelect.innerHTML = '<option value="">-- 选择一个方案 --</option>';
                    profileNames.sort().forEach(name => {
                        const option = document.createElement('option');
                        option.value = name;
                        option.textContent = name;
                        profileSelect.appendChild(option);
                    });

                const reader = new FileReader();
                reader.onload = (e) => {
                    try {
                        const importedProfiles = JSON.parse(e.target.result);
                        if (typeof importedProfiles !== 'object' || importedProfiles === null) { throw new Error('无效的文件格式。'); }
                        let currentProfiles = getProfiles();
                        let importedCount = 0, skippedCount = 0;
                        for (const name in importedProfiles) {
                            if (currentProfiles[name]) {
                                if (confirm(`已存在同名方案 "${name}"，是否要覆盖它？`)) { currentProfiles[name] = importedProfiles[name]; importedCount++; }
                                else { skippedCount++; }
                            } else { currentProfiles[name] = importedProfiles[name]; importedCount++; }
                        }

                        populateDropdown();
                        showNotification(`成功导入 ${importedCount} 个方案，跳过 ${skippedCount} 个。`);

                        showNotification('导入失败！请确保文件格式正确。', 'var(--error-color)');
                        console.error("Import error:", error);

                        importFileInput.value = '';
                    }
