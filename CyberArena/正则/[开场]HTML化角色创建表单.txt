<script>
document.addEventListener('DOMContentLoaded', () => {
    const form = document.getElementById('creator-form');
    const generateBtn = document.getElementById('generate-btn');
    const resetFormBtn = document.getElementById('reset-form-btn');
    const notification = document.getElementById('notification');
    const initialButtonHTML = generateBtn.innerHTML;
    const allInputs = Array.from(form.querySelectorAll('input, textarea, select'));

    const profileSelect = document.getElementById('profile-select');
    const loadProfileBtn = document.getElementById('load-profile-btn');
    const saveProfileBtn = document.getElementById('save-profile-btn');
    const deleteProfileBtn = document.getElementById('delete-profile-btn');
    const exportProfilesBtn = document.getElementById('export-profiles-btn');
    const importProfilesBtn = document.getElementById('import-profiles-btn');
    const importFileInput = document.getElementById('import-file-input');

    const AUTOSAVE_KEY = 'cyberarena_AutoSave_v1';
    const PROFILE_STORAGE_KEY = 'cyberarena_ProfileManager_v1';
    let saveTimeout;

    // --- 文本框高度自适应 ---
    const textareas = document.querySelectorAll('textarea');
    const autoResizeTextarea = (ta) => {
        ta.style.height = 'auto';
        ta.style.height = ta.scrollHeight + 'px';
    };
    textareas.forEach(ta => {
        ta.addEventListener('input', () => { autoResizeTextarea(ta); debouncedAutoSave(); });
    });

    const showNotification = (message, color = 'var(--cyan)') => {
        notification.textContent = message;
        notification.style.color = color;
        notification.style.opacity = '1';
        clearTimeout(notification.timeoutId);
        notification.timeoutId = setTimeout(() => { notification.style.opacity = '0'; }, 3500);
    };

    const getCurrentFormData = () => {
        const data = {};
        allInputs.forEach(input => {
            if (input.id) data[input.id] = input.value;
        });
        return data;
    };

    const populateForm = (data) => {
        if (!data) return;
        allInputs.forEach(input => {
            const key = input.id;
            if (data[key] !== undefined) {
                input.value = data[key];
            } else if (input.type === 'text' || input.tagName === 'TEXTAREA') {
                input.value = '';
            }
        });
        setTimeout(() => { textareas.forEach(ta => autoResizeTextarea(ta)); }, 10);
    };

    const saveAutoSave = () => localStorage.setItem(AUTOSAVE_KEY, JSON.stringify(getCurrentFormData()));
    const debouncedAutoSave = () => { clearTimeout(saveTimeout); saveTimeout = setTimeout(saveAutoSave, 500); };

    const loadAutoSave = () => {
        const autoSavedData = localStorage.getItem(AUTOSAVE_KEY);
        if (autoSavedData) {
            populateForm(JSON.parse(autoSavedData));
        }
    };

    const getProfiles = () => JSON.parse(localStorage.getItem(PROFILE_STORAGE_KEY)) || {};
    const saveProfiles = (profiles) => localStorage.setItem(PROFILE_STORAGE_KEY, JSON.stringify(profiles));

    const populateDropdown = () => {
        const profiles = getProfiles();
        const profileNames = Object.keys(profiles);
        profileSelect.innerHTML = '';
        if (profileNames.length === 0) {
            profileSelect.innerHTML = '<option value="">-- 尚无已存方案 --</option>';
            loadProfileBtn.disabled = true;
            deleteProfileBtn.disabled = true;
            exportProfilesBtn.disabled = true;
        } else {
            profileSelect.innerHTML = '<option value="">-- 选择一个方案 --</option>';
            profileNames.sort().forEach(name => {
                const option = document.createElement('option');
                option.value = name;
                option.textContent = name;
                profileSelect.appendChild(option);
            });
            loadProfileBtn.disabled = false;
            deleteProfileBtn.disabled = false;
            exportProfilesBtn.disabled = false;
        }
    };

    form.addEventListener('input', debouncedAutoSave);

    saveProfileBtn.addEventListener('click', () => {
        const defaultName = document.getElementById('char-name').value || '未命名方案';
        const profileName = prompt('请输入方案名称：', defaultName);
        if (profileName && profileName.trim() !== '') {
            const profiles = getProfiles();
            profiles[profileName.trim()] = getCurrentFormData();
            saveProfiles(profiles);
            populateDropdown();
            profileSelect.value = profileName.trim();
            showNotification(`方案 "${profileName.trim()}" 已保存！`);
        }
    });

    loadProfileBtn.addEventListener('click', () => {
        const selectedName = profileSelect.value;
        if (selectedName) {
            const profiles = getProfiles();
            if (profiles[selectedName]) {
                populateForm(profiles[selectedName]);
                debouncedAutoSave();
                showNotification(`已读取方案 "${selectedName}"。`);
            }
        } else {
            showNotification('请先从下拉栏中选择一个方案。', 'var(--pink)');
        }
    });

    deleteProfileBtn.addEventListener('click', () => {
        const selectedName = profileSelect.value;
        if (selectedName) {
            if (confirm(`确定要永久删除方案 "${selectedName}" 吗？此操作无法撤销。`)) {
                const profiles = getProfiles();
                delete profiles[selectedName];
                saveProfiles(profiles);
                populateDropdown();
                showNotification(`方案 "${selectedName}" 已被删除。`, 'var(--pink)');
            }
        } else {
            showNotification('请先从下拉栏中选择一个要删除的方案。', 'var(--pink)');
        }
    });

    exportProfilesBtn.addEventListener('click', () => {
        const profiles = getProfiles();
        if (Object.keys(profiles).length === 0) return;
        const dataStr = JSON.stringify(profiles, null, 2);
        const dataBlob = new Blob([dataStr], {type: 'application/json'});
        const url = URL.createObjectURL(dataBlob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `cyberarena-profiles-${Date.now()}.json`;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
        showNotification('所有方案已成功导出！');
    });

    importProfilesBtn.addEventListener('click', () => importFileInput.click());

    importFileInput.addEventListener('change', (event) => {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = (e) => {
            try {
                const importedProfiles = JSON.parse(e.target.result);
                if (typeof importedProfiles !== 'object' || importedProfiles === null) { throw new Error('无效的文件格式。'); }
                let currentProfiles = getProfiles();
                let importedCount = 0, skippedCount = 0;
                for (const name in importedProfiles) {
                    if (currentProfiles[name]) {
                        if (confirm(`已存在同名方案 "${name}"，是否要覆盖它？`)) { currentProfiles[name] = importedProfiles[name]; importedCount++; }
                        else { skippedCount++; }
                    } else { currentProfiles[name] = importedProfiles[name]; importedCount++; }
                }
                saveProfiles(currentProfiles);
                populateDropdown();
                showNotification(`成功导入 ${importedCount} 个方案，跳过 ${skippedCount} 个。`);
            } catch (error) {
                showNotification('导入失败！请确保文件格式正确。', 'var(--pink)');
                console.error("Import error:", error);
            } finally {
                importFileInput.value = '';
            }
        };
        reader.readAsText(file);
    });

    resetFormBtn.addEventListener('click', () => {
        if (confirm('确定要清空当前所有填写的内容，并将表单重置为默认状态吗？')) {
            form.reset();
            debouncedAutoSave();
            showNotification('表单已重置为默认状态。', 'var(--cyan)');
        }
    });

    generateBtn.addEventListener('click', () => {
        saveAutoSave();
        generateBtn.disabled = true;
        generateBtn.innerHTML = '<span>正在报名...</span>';
        notification.style.opacity = '0';

        const data = getCurrentFormData();
        const formatValue = (value, placeholder = '[AI请填充]') => {
            const trimmedValue = (value || "").trim();
            return trimmedValue === '' ? placeholder : `"${trimmedValue}"`;
        };

        const yamlString = `<init_data>
\`\`\`yaml
# AI任务指令：解析以下用户创角档案，生成开场白并初始化所有变量。
# 对于值为 "[AI请填充]" 的字段，你【必须】基于已有设定进行富有想象力的、合理的填充。

创角档案:
  基本信息:
    姓名: ${formatValue(data['char-name'], '')}
    年龄: ${formatValue(data['char-age'])}
    种族: ${formatValue(data['char-race'])}
    擂台代号: ${formatValue(data['char-alias'])}
  身份与性格:
    身份: ${formatValue(data['char-identity'])}
    性格特质: ${formatValue(data['char-traits'])}
  简要外貌:
    外貌关键词: ${formatValue(data['appearance-keywords'])}
    肉体关键词: ${formatValue(data['body-keywords'])}
  人际关系: ${formatValue(data['char-relationships'], '无')}
\`\`\`
</init_data>`;

        setTimeout(() => {
            try {
                const inputArea = window.parent.document.querySelector('#send_textarea');
                const sendButton = window.parent.document.querySelector('#send_but');
                if (inputArea && sendButton) {
                    inputArea.value = yamlString;
                    inputArea.dispatchEvent(new Event('input', { bubbles: true }));
                    sendButton.click();
                    showNotification('报名成功！等待AI回应...');
                    generateBtn.innerHTML = '<span>报名已发送</span>';
                } else { throw new Error('SillyTavern UI not found.'); }
            } catch (error) {
                console.error('Error sending message:', error);
                generateBtn.innerHTML = '<span>发送失败</span>';
                navigator.clipboard.writeText(yamlString).then(() => {
                    showNotification('自动发送失败！已复制到剪贴板。', 'var(--pink)');
                });
            }
            setTimeout(() => {
                generateBtn.disabled = false;
                generateBtn.innerHTML = initialButtonHTML;
            }, 5000);
        }, 500);
    });

    // 初始化
    populateDropdown();
    loadAutoSave();
});
</script>
</body>
</html>
