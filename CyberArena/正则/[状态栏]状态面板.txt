function updateDOM(fromNode, toNode) {
  if (!fromNode || !toNode) return;
  const fa = fromNode.attributes, ta = toNode.attributes;
  for (const a of ta) { if (fromNode.getAttribute(a.name) !== a.value) fromNode.setAttribute(a.name, a.value); }
  for (const a of fa) { if (!toNode.hasAttribute(a.name)) fromNode.removeAttribute(a.name); }
  const fc = Array.from(fromNode.childNodes), tc = Array.from(toNode.childNodes);
  for (let i = 0; i < Math.max(fc.length, tc.length); i++) {
    const f = fc[i], t = tc[i];
    if (!t) { fromNode.removeChild(f); }
    else if (!f) { fromNode.appendChild(t.cloneNode(true)); }
    else if (f.nodeType !== t.nodeType || f.tagName !== t.tagName) { fromNode.replaceChild(t.cloneNode(true), f); }
    else if (f.nodeType === Node.TEXT_NODE) { if (f.textContent !== t.textContent) f.textContent = t.textContent; }
    else if (f.nodeType === Node.ELEMENT_NODE) { updateDOM(f, t); }
  }
}

const renderTabRoster = function(data) {
  var roster = data.记录 || {};
  var fighters = roster.性斗士名录 || {};
  var npcs = roster.一般NPC名录 || {};

  var allEntries = [];
  Object.keys(fighters).forEach(function(k) {
    var f = fighters[k];
    allEntries.push({ key: k, name: f.姓名 || k, type: 'fighter', relation: safeStr(f.与主角关系), favor: safeNum(f.好感度), rank: safeStr(f.擂台段位), data: f });
  });
  Object.keys(npcs).forEach(function(k) {
    var n = npcs[k];
    allEntries.push({ key: k, name: k, type: 'npc', relation: safeStr(n.身份描述), favor: safeNum(n.好感度), data: n });
  });

  var listHTML = '';
  if (allEntries.length === 0) {
    listHTML = '<div class="empty-hint">暂无记录</div>';
  } else {
    allEntries.forEach(function(e) {
      var cls = 'roster-item ' + e.type;
      if (e.key === selectedRosterKey) cls += ' sel';
      listHTML += '<div class="' + cls + '" data-key="' + e.key + '" data-type="' + e.type + '">';
      listHTML += '<div class="r-name">' + (e.type === 'fighter' ? '🔴 ' : '🟡 ') + e.name + '</div>';
      listHTML += '<div class="r-sub">' + e.relation + ' · 好感 ' + e.favor + (e.type === 'fighter' ? ' · ' + e.rank : '') + '</div>';
      listHTML += '</div>';
    });
  }

  var detailHTML = '<div class="empty-hint">选择左侧人物查看详情</div>';
  if (selectedRosterKey) {
    var sel = allEntries.find(function(e) { return e.key === selectedRosterKey; });
    if (sel) detailHTML = showRosterDetail(sel.key, sel.type, sel.data);
  }

  return '<div class="roster-layout"><div class="roster-list">' + listHTML + '</div><div class="roster-detail">' + detailHTML + '</div></div>';
};

const renderTabHistory = function(data) {
  var history = data.记录?.历史性斗记录 || [];
  if (history.length === 0) return '<div class="history-wrap"><div class="empty-hint">暂无性斗记录</div></div>';

  var sorted = history.slice().reverse();
  var html = '<div class="history-wrap">';
  sorted.forEach(function(h) {
    var result = safeStr(h.结果);
    var icon = '🤝', iconCls = 'draw';
    if (result === '胜') { icon = '🏆'; iconCls = 'win'; }
    else if (result === '负') { icon = '💀'; iconCls = 'lose'; }
    var ptsDiff = safeNum(h.赛后积分) - safeNum(h.赛前积分);
    var ptsCls = ptsDiff > 0 ? 'up' : (ptsDiff < 0 ? 'down' : '');
    var popDiff = safeNum(h.人气变化);
    var popCls = popDiff > 0 ? 'up' : (popDiff < 0 ? 'down' : '');
    var credDiff = safeNum(h.信用点变化);
    var credCls = credDiff > 0 ? 'up' : (credDiff < 0 ? 'down' : '');

    html += '<div class="hist-entry">';
    html += '<span class="hist-icon ' + iconCls + '">' + icon + '</span>';
    html += '<span class="hist-main">' + safeStr(h.性斗日期) + ' vs ' + safeStr(h.对手名字) + '<span class="hist-rank">[' + safeStr(h.对手段位) + ']</span></span>';
    html += '<div class="hist-sub">';
    html += '结果：<span>' + result + '</span> | 积分 <span>' + safeNum(h.赛前积分) + ' → ' + safeNum(h.赛后积分) + '</span> <span class="' + ptsCls + '">(' + sign(ptsDiff) + ')</span> | ';
    html += '人气 <span class="' + popCls + '">' + sign(popDiff) + '</span> | ⊕ <span class="credits ' + credCls + '">' + sign(credDiff) + '</span>';
    html += '</div></div>';
  });
  html += '</div>';
  return html;
};

const renderPanelWithData = function(stat_data) {
  var container = document.getElementById('status-card-container');
  if (!container) return;
  try {
    var newHTML = renderStatusPanel(stat_data);
    var tempDiv = document.createElement('div');
    tempDiv.innerHTML = newHTML;
    var newContent = tempDiv.firstChild;
    if (newContent && newContent.querySelector) {
      var newActiveButton = newContent.querySelector('[data-target="' + lastActiveTabId + '"]');
      if (newActiveButton) newActiveButton.classList.add('active');
      var newActivePanel = newContent.querySelector('#' + lastActiveTabId);
      if (newActivePanel) newActivePanel.classList.add('active');
    }
    if (container.firstChild) { updateDOM(container.firstChild, newContent); }
    else { if (newContent) container.appendChild(newContent); }
  } catch (error) { console.error('Status Panel Error:', error); container.innerHTML = '<div style="color:var(--pink);">渲染出错: ' + error.message + '</div>'; }

  // Re-attach item popup handler
  setTimeout(function() {
    var itemsBtn = document.getElementById('items-btn');
    if (itemsBtn) {
      itemsBtn.addEventListener('click', function(e) {
        e.stopPropagation();
        var popContent = document.getElementById('item-pop-content');
        var items = (stat_data['主角'] && stat_data['主角']['道具栏']) || {};

document.getElementById('status-card-container').addEventListener('click', function(event) {
  var button = event.target.closest('.tab-btn');
  if (button && !button.classList.contains('active')) {
    var targetPanelId = button.getAttribute('data-target');
    var container = document.getElementById('status-card-container');
    container.querySelectorAll('.tab-btn').forEach(function(btn) { btn.classList.remove('active'); });
    container.querySelectorAll('.tab-panel').forEach(function(panel) { panel.classList.remove('active'); });
    button.classList.add('active');
    var targetPanel = container.querySelector('#' + targetPanelId);
    if (targetPanel) targetPanel.classList.add('active');
    lastActiveTabId = targetPanelId;
  }

  // Roster item click
  var rosterItem = event.target.closest('.roster-item');
  if (rosterItem) {
    var container = document.getElementById('status-card-container');
    container.querySelectorAll('.roster-item').forEach(function(r) { r.classList.remove('sel'); });
    rosterItem.classList.add('sel');
    selectedRosterKey = rosterItem.getAttribute('data-key');
    var type = rosterItem.getAttribute('data-type');
    var roster = window._lastStatData?.记录 || {};
    var d = type === 'fighter' ? (roster.性斗士名录 || {})[selectedRosterKey] : (roster.一般NPC名录 || {})[selectedRosterKey];
    if (d) {
      var detailEl = container.querySelector('.roster-detail');
      if (detailEl) detailEl.innerHTML = showRosterDetail(selectedRosterKey, type, d);
    }
  }

  // Quick actions
  var actionEl = event.target.closest('[data-action]');
  if (actionEl) {
    event.preventDefault(); event.stopPropagation();
    var text = actionEl.getAttribute('data-action');
    var textarea = window.parent.document.querySelector('#send_textarea') || document.querySelector('textarea');
    if (textarea) {
      textarea.value = text;
      textarea.dispatchEvent(new Event('input', { bubbles: true }));
      textarea.focus();
    }
  }
});

function getMvuInstance() {
  try { return window.Mvu || (window.parent && window.parent.Mvu) || (window.top && window.top.Mvu) || null; }
  catch(e) { return null; }
}

const init = function() {
  var Mvu = getMvuInstance();
  if (Mvu && Mvu.events) {
    document.addEventListener(Mvu.events.VARIABLE_UPDATE_ENDED, function(e) {
      if (e.detail && e.detail.stat_data) {
        window._lastStatData = e.detail.stat_data;
        renderPanelWithData(e.detail.stat_data);
      }
    });
    (async function() {
      try {
        var data;
        if (typeof getChatMessages !== 'undefined') {
          var msgs = await getChatMessages(getCurrentMessageId());
          data = msgs[0]?.data?.stat_data || Mvu.getMvuData({type:'message', message_id:-1})?.stat_data;
        } else {
          data = Mvu.getMvuData({type:'message', message_id:-1})?.stat_data;
        }
        if (data) { window._lastStatData = data; renderPanelWithData(data); }
      } catch(e) {}
    })();
  } else {
    setTimeout(init, 500);
  }
};

if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', init); } else { init(); }
