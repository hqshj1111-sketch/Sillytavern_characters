    const isDepleted = (value) => {
        if (typeof value === 'number') return value <= 0;
        if (typeof value === 'boolean') return value === false;

    const classifyItem = (key, value) => {
        if (isDepleted(value)) return 'depleted';
        if (DECORATIVE_KEYWORDS.test(key)) return 'decorative';
        // ← 扩充：增加 情报_/数据_/档案_ 前缀 + 核心/模块/参数/存储体 等关键词
        if (/^(关键|任务|线索|剧情|情报|数据|档案|记录)_/i.test(key)

        if (/^(武器|装备|防具|武装|变身|饰品|戒指|项链|护符|魔杖)_/i.test(key)

            return 'equipment';
        }
        if (typeof value === 'number' || /^\d+$/.test(String(value).trim())) return 'consumable';

    const renderKeys = (arr) => arr.length === 0 ? null : `["${arr.join('", "')}"]`;

        const keys = renderKeys(arr);
        if (!keys) return '';

    checks.push('事件本身是否已 remove？');
    const checkLines = checks.map(c => `      - "${c}"`).join('\n');
