    function updateDOM(fromNode, toNode) {
        if (!fromNode || !toNode) return;

        const fromAttrs = fromNode.attributes;
        const toAttrs = toNode.attributes;
        for (const attr of toAttrs) {
            if (fromNode.getAttribute(attr.name) !== attr.value) {
                fromNode.setAttribute(attr.name, attr.value);
            }
        }
        for (const attr of fromAttrs) {
            if (!toNode.hasAttribute(attr.name)) {
                fromNode.removeAttribute(attr.name);
            }
        }

        const fromChildren = Array.from(fromNode.childNodes);
        const toChildren = Array.from(toNode.childNodes);
        const maxLen = Math.max(fromChildren.length, toChildren.length);

        for (let i = 0; i < maxLen; i++) {
            const fromChild = fromChildren[i];
            const toChild = toChildren[i];
            if (!toChild) {
                fromNode.removeChild(fromChild);
            } else if (!fromChild) {
                fromNode.appendChild(toChild.cloneNode(true));
            } else if (fromChild.nodeType !== toChild.nodeType || fromChild.tagName !== toChild.tagName) {
                fromNode.replaceChild(toChild.cloneNode(true), fromChild);
            } else if (fromChild.nodeType === Node.TEXT_NODE) {
                if (fromChild.textContent !== toChild.textContent) {
                    fromChild.textContent = toChild.textContent;
                }
            } else if (fromChild.nodeType === Node.ELEMENT_NODE) {
                updateDOM(fromChild, toChild);
            }
        }
    }

            portal.className = 'custom-tooltip-portal';
            document.body.appendChild(portal);
        }

        const showTip = (target) => {
            const tpl = target.querySelector('.custom-tooltip');
            if (!tpl || !portal) return;

            portal.innerHTML = tpl.innerHTML;
            portal.classList.add('active');

            const targetRect = target.getBoundingClientRect();
            const tipRect = portal.getBoundingClientRect();

            let top = targetRect.top + (targetRect.height / 2) - (tipRect.height / 2);
            let left = targetRect.right + 12;

            // 边缘避让算法：若右侧超出屏幕，则翻转至目标元素左侧
            if (left + tipRect.width > window.innerWidth - 10) {
                left = targetRect.left - tipRect.width - 12;
            }
            if (left < 10) left = 10;

            // 垂直方向上下边界贴靠
            if (top + tipRect.height > window.innerHeight - 10) {
                top = window.innerHeight - tipRect.height - 10;
            }
            if (top < 10) top = 10;

            portal.style.top = `${top}px`;
            portal.style.left = `${left}px`;
        };

        const hideTip = () => {
            if (portal) {
                portal.classList.remove('active');
            }
        };

        document.addEventListener('mouseover', (e) => {
            const trigger = e.target.closest('.has-tooltip');
            if (trigger) showTip(trigger);
        });

        document.addEventListener('mouseout', (e) => {
            const trigger = e.target.closest('.has-tooltip');
            if (trigger) hideTip();
        });

        window.addEventListener('scroll', hideTip, true);
    };

                    console.warn("[PC Status] Initial fetch failed.", e);
                }
            })();
            const container = document.getElementById('status-card-container');
            if (container) {
                container.addEventListener('click', (event) => {
                    const button = event.target.closest('.tab-btn');
                    if (!button || button.classList.contains('active')) return;
                    const targetPanelId = button.dataset.target;
                    const targetPanel = container.querySelector(`#${targetPanelId}`);
                    if (!targetPanel) return;
                    container.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
                    container.querySelectorAll('.tab-content').forEach(panel => panel.classList.remove('active'));
                    button.classList.add('active');
                    targetPanel.classList.add('active');
                    lastActiveTabId = targetPanelId;
                });
            }
        } else {
            setTimeout(init, 500);
        }
    };

    window.gm_update = async (path, value, mode = 'set') => {
        try {
            const Mvu = window.Mvu || window.parent?.Mvu || window.top?.Mvu;
            if (!Mvu) return;
            const mvuData = Mvu.getMvuData({ type: 'message', message_id: -1 });
            if (!mvuData) return;
            if (typeof _ !== 'undefined' && _.set) {
                if (mode === 'remove') _.unset(mvuData, path);
                else if (mode === 'add') {
                    const old = _.get(mvuData, path) || 0;
                    _.set(mvuData, path, Number(old) + Number(value));
                } else {
                    _.set(mvuData, path, value);
                }
            }
            await Mvu.replaceMvuData(mvuData, { type: 'message', message_id: -1 });
            const cleanData = JSON.parse(JSON.stringify(mvuData.stat_data || mvuData));
            document.dispatchEvent(new CustomEvent(Mvu.events.VARIABLE_UPDATE_ENDED, { detail: { stat_data: cleanData }, bubbles: true }));
            setTimeout(() => {
                try { renderGM && renderGM(); } catch(e) {}
            }, 50);
        } catch (e) {
            console.error("[GM Update]", e);
        }
    };
