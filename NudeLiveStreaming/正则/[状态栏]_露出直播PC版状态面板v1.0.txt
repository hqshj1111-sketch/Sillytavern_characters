function updClock(){
  const y=getValue(globalData,'进程.当前年份'),mo=getValue(globalData,'进程.当前月份'),d=getValue(globalData,'进程.当前日份'),h=getValue(globalData,'进程.当前小时'),mi=getValue(globalData,'进程.当前分钟'),w=getValue(globalData,'进程.当前周几');
  const mNames=['1月','2月','3月','4月','5月','6月','7月','8月','9月','10月','11月','12月'];
  let season='';const mn=mo||10;if(mn>=3&&mn<=5)season='春季';else if(mn>=6&&mn<=8)season='夏季';else if(mn>=9&&mn<=11)season='秋季';else season='冬季';
  const wNames=['周一','周二','周三','周四','周五','周六','周日'];
  document.getElementById('hdr-date').textContent=(y||2025)+'-'+String(mn).padStart(2,'0')+'-'+String(d||15).padStart(2,'0')+' '+season+' '+(wNames[w-1]||'周三');
  document.getElementById('hdr-time').textContent=String(h||19).padStart(2,'0')+':'+String(mi||0).padStart(2,'0');
  // Weather
  const weather=getValue(globalData,'进程.当前天气')||'晴';
  const wh=document.getElementById('hdr-weather');
  if(weather==='晴')wh.innerHTML=IC.sun+' <span>'+weather+'</span>';
  else if(weather.includes('雨'))wh.innerHTML=IC.cloudRain+' <span>'+weather+'</span>';
  else wh.innerHTML=IC.cloud+' <span>'+weather+'</span>';
  // Location
  const area=getValue(globalData,'主角.当前状态.当前区域')||'';const loc=getValue(globalData,'主角.当前状态.当前地点')||'';
  document.getElementById('hdr-loc').textContent=area+(loc?' · '+loc:'');
  // Update timer
  U._updateTimer();
}

  }

const init=async()=>{
  const Mvu=getMvuInstance();
  if(Mvu&&Mvu.events){
    document.addEventListener(Mvu.events.VARIABLE_UPDATE_ENDED,(e)=>{
      if(e.detail&&e.detail.stat_data){
        globalData=e.detail.stat_data;
        renderApp();
      }
    });
    const wrapper=Mvu.getMvuData({type:'message',message_id:-1});
    if(wrapper){
      globalData=wrapper.stat_data||wrapper;
      renderApp();
    }
  }else{
    // Standalone: use default data
    globalData=JSON.parse(JSON.stringify(DEFAULT_STAT));
    renderApp();
    setTimeout(init,1000);
  }
};

  document.getElementById('idle-avatar-text').textContent=(name||'蓮')[0];
  updClock();
  // Begin MVU connection
  init();
};

const getDeep = (obj, path, def) => path.split('.').reduce((acc, part) => acc && acc[part], obj) ?? def;

const showToast = (msg, selector) => {
  const el = document.querySelector(selector);
  if (el) {
    el.innerText = msg;
    el.style.opacity = '1';
    setTimeout(() => { el.style.opacity = '0'; }, 2000);
  }
};

const setInp = (id, val) => {
  const el = document.getElementById(id);
  if (el && document.activeElement !== el) el.value = (typeof val === 'number') ? Math.round(val) : val;
};

const setTxt = (id, val) => {
  const el = document.getElementById(id);
  if (el) el.innerText = val;
};

// ------------------- 面板显示/隐藏 -------------------
function toggleGM() {
  const panel = document.getElementById('gm-panel');
  if (!panel) return;
  if (!panel.classList.contains('show')) {
    panel.style.display = 'flex';
    requestAnimationFrame(() => {
      panel.classList.add('show');
      renderGM(); // 打开时刷新所有数据
      if (_currentMvuData) {
        _gmSnapshot = JSON.parse(JSON.stringify(_currentMvuData));
        showToast('快照已保存，可撤销', '#save-status');
      }
    });
  } else {
    panel.classList.remove('show');
    setTimeout(() => { panel.style.display = 'none'; }, 250);
  }
}

async function renderGM() {
  try {
    const Mvu = getMvuInstance();
    if (!Mvu) return;
    const wrapper = Mvu.getMvuData({ type: 'message', message_id: -1 });
    if (!wrapper) return;
    _currentMvuData = wrapper.stat_data || wrapper;
    const d = _currentMvuData;

    // --- 阶段显示 ---
    const phase = getDeep(d, '进程.阶段', '日常阶段');
    setTxt('current-phase-display', phase);
    setInp('phase-selector', phase);

    // --- 核心数值 ---
    setInp('inp-pleasure', getDeep(d, '主角.当前状态.快感值', 0));
    setInp('inp-desire', getDeep(d, '主角.当前状态.性欲值', 0));
    setInp('inp-shame', getDeep(d, '主角.当前状态.羞耻度', 100));
    setInp('inp-lewd', getDeep(d, '主角.当前状态.淫乱度', 0));
    setInp('inp-points', getDeep(d, '主角.主播信息.直播点数', 0));
    setInp('inp-fans', getDeep(d, '主角.主播信息.粉丝数', 0));
    setInp('inp-stream-count', getDeep(d, '主角.主播信息.直播总次数', 0));

    // --- 直播状态 ---
    const isStreaming = getDeep(d, '直播状态.是否正在直播', false);
    document.getElementById('live-status-display').innerText = isStreaming ? '直播中' : '未开播';
    setInp('inp-viewers', getDeep(d, '直播状态.当前直播间人数', 0));
    setInp('inp-heat', getDeep(d, '直播状态.当前直播间热度', 0));

    // --- 待处理事件队列 ---
    const events = getDeep(d, '系统状态.待处理事件', {});
    const eventKeys = Object.keys(events).filter(k => k !== '$meta' && k !== '$__META_EXTENSIBLE__$');
    const eventContainer = document.getElementById('event-list-container');
    if (eventKeys.length > 0) {
        eventContainer.innerHTML = eventKeys.map(k => {
        const val = events[k];
        let displayVal = '';
        if (typeof val === 'string') {
          displayVal = val;
        } else if (typeof val === 'boolean') {
          displayVal = val.toString();
        } else if (typeof val === 'object') {
          displayVal = JSON.stringify(val, null, 1);
        } else if (val !== undefined) {
          displayVal = String(val);
        }
        return `<div class="event-item">
          <div style="flex:1; display:flex; justify-content:space-between; align-items:center; gap:8px;">
            <span style="font-weight:600; color:#c0caf5;">${k}</span>

          </div>
          <i class="fa-solid fa-xmark del" onclick="gm_update('stat_data.系统状态.待处理事件.${k.replace(/'/g, "\\'")}', null, 'remove')"></i>
        </div>`;
      }).join('');
    } else {
      eventContainer.innerHTML = '<div style="text-align:center; opacity:0.5; padding:4px;">(空)</div>';
    }

    // 动态更新“对象”候选列表
    const targetList = document.getElementById('act-target-list');
    if (targetList) {
        const defaultTargets = ['自己', '路人', '陌生人', '汐见真帆', '黑崎海斗', '冰室纱耶', '风见迅'];

// ------------------- 核心变量修改函数 -------------------
async function gm_update(path, value, mode = 'set') {
  try {
    const Mvu = getMvuInstance();
    if (!Mvu) return showToast('未连接到MVU系统', '#save-status');
    const mvuData = Mvu.getMvuData({ type: 'message', message_id: -1 });
    if (!mvuData) return;

    if (typeof _ !== 'undefined' && _.set) {
      if (mode === 'remove') {
        _.unset(mvuData, path);
      } else if (mode === 'add') {
        const old = _.get(mvuData, path) || 0;
        _.set(mvuData, path, Number(old) + Number(value));
      } else {
        _.set(mvuData, path, value);
      }
    }

    await Mvu.replaceMvuData(mvuData, { type: 'message', message_id: -1 });
    const cleanData = JSON.parse(JSON.stringify(mvuData.stat_data || mvuData));
    document.dispatchEvent(new CustomEvent(Mvu.events.VARIABLE_UPDATE_ENDED, { detail: { stat_data: cleanData }, bubbles: true }));
    setTimeout(renderGM, 50);
    showToast('变量已更新', '#save-status');
  } catch (e) {
    console.error('[GM Update]', e);
    showToast('更新失败', '#save-status');
  }
}

async function _getCurrentActs() {
    const Mvu = getMvuInstance();
    if (!Mvu) throw new Error('MVU未连接');
    const mvuData = Mvu.getMvuData({ type: 'message', message_id: -1 });
    if (!mvuData) throw new Error('无法获取MVU数据');
    let acts = _.get(mvuData, 'stat_data.主角.当前状态.当前性行为') || [];
    if (!Array.isArray(acts)) acts = [];
    return acts;
}

async function addSexAct() {
    const target = document.getElementById('new-act-target')?.value.trim();
    if (!target) return alert('请填写对象名称');
    const gender = document.getElementById('new-act-gender')?.value || '';
    const part   = document.getElementById('new-act-part')?.value || '';
    const action = document.getElementById('new-act-action')?.value || '';
    if (!part || !action) return alert('请选择部位和行为');

    const newAct = { 对象: target, 部位: part, 行为: action };
    if (gender) newAct.性别 = gender;

    try {
        const acts = await _getCurrentActs();
        acts.push(newAct);
        await gm_update('stat_data.主角.当前状态.当前性行为', acts);
        // 清空输入框
        ['new-act-target', 'new-act-gender', 'new-act-part', 'new-act-action'].forEach(id => {
            const el = document.getElementById(id);
            if (el) el.value = '';
        });
    } catch (e) { console.error('[addSexAct]', e); }
}

async function removeSexAct(index) {
    try {
        const acts = await _getCurrentActs();
        if (index < 0 || index >= acts.length) return;
        if (!confirm('确定删除这个性行为？')) return;
        acts.splice(index, 1);
        await gm_update('stat_data.主角.当前状态.当前性行为', acts);
    } catch (e) { console.error('[removeSexAct]', e); }
}

async function editSexAct(index) {
    try {
        const acts = await _getCurrentActs();
        if (index < 0 || index >= acts.length) return;
        const act = acts[index];
        const newJson = prompt('编辑性行为 (JSON格式)', JSON.stringify(act, null, 2));
        if (!newJson) return;
        const parsed = JSON.parse(newJson);
        acts[index] = parsed;
        await gm_update('stat_data.主角.当前状态.当前性行为', acts);
    } catch (e) { console.error('[editSexAct]', e); }
}

// 添加自定义事件到待处理队列
async function addCustomEvent() {
    const nameInput = document.getElementById('new-event-input');
    const valueInput = document.getElementById('new-event-value');
    const eventName = nameInput.value.trim();
    if (!eventName) return;

    let eventValue = true; // 默认布尔值
    const rawVal = valueInput.value.trim();
    if (rawVal !== '') {
        // 尝试解析为布尔/数字/JSON，失败则保留字符串
        if (rawVal === 'true') eventValue = true;
        else if (rawVal === 'false') eventValue = false;
        else if (!isNaN(rawVal)) eventValue = Number(rawVal);
        else {
            try { eventValue = JSON.parse(rawVal); } catch (e) { eventValue = rawVal; }
        }
    }

    await gm_update(`stat_data.系统状态.待处理事件.${eventName}`, eventValue);
    nameInput.value = '';
    valueInput.value = '';
    showToast(`已添加事件: ${eventName}`, '#save-status');
}
