<script>
document.addEventListener('DOMContentLoaded',()=>{
  const form=document.getElementById('creator-form');
  const generateBtn=document.getElementById('generate-btn');
  const resetBtn=document.getElementById('reset-btn');
  const notification=document.getElementById('notification');
  const allInputs=Array.from(form.querySelectorAll('input[type=text], textarea'));
  const profileSelect=document.getElementById('profile-select');
  const loadBtn=document.getElementById('load-btn');
  const saveBtn=document.getElementById('save-btn');
  const deleteBtn=document.getElementById('delete-btn');
  const exportBtn=document.getElementById('export-btn');
  const importBtn=document.getElementById('import-btn');
  const importFile=document.getElementById('import-file');
  const initialBtnHTML=generateBtn.innerHTML;

  const AUTOSAVE_KEY='nls_charCreate_autosave_v1';
  const PROFILE_KEY='nls_charCreate_profiles_v1';
  let saveTimeout;

  const notify=(msg,color='#4aab73')=>{
    notification.textContent=msg;notification.style.color=color;notification.style.opacity='1';
    clearTimeout(notification._tid);notification._tid=setTimeout(()=>{notification.style.opacity='0';},3500);
  };

  const getFormData=()=>{
    const data={};
    allInputs.forEach(inp=>{if(inp.id)data[inp.id]=inp.value;});
    return data;
  };

  const populateForm=(data)=>{
    if(!data)return;
    allInputs.forEach(inp=>{const key=inp.id;if(data[key]!==undefined)inp.value=data[key];else if(inp.tagName!=='SELECT')inp.value='';});
  };

  const autoSave=()=>localStorage.setItem(AUTOSAVE_KEY,JSON.stringify(getFormData()));
  const debouncedSave=()=>{clearTimeout(saveTimeout);saveTimeout=setTimeout(autoSave,500);};

  const getProfiles=()=>JSON.parse(localStorage.getItem(PROFILE_KEY))||{};
  const saveProfiles=(p)=>localStorage.setItem(PROFILE_KEY,JSON.stringify(p));

  const refreshDropdown=()=>{
    const profiles=getProfiles();const names=Object.keys(profiles);
    profileSelect.innerHTML='';
    if(names.length===0){
      profileSelect.innerHTML='<option value="">-- 尚无已存方案 --</option>';
      loadBtn.disabled=true;deleteBtn.disabled=true;exportBtn.disabled=true;
    }else{
      profileSelect.innerHTML='<option value="">-- 选择方案 --</option>';
      names.sort().forEach(n=>{const o=document.createElement('option');o.value=n;o.textContent=n;profileSelect.appendChild(o);});
      loadBtn.disabled=false;deleteBtn.disabled=false;exportBtn.disabled=false;
    }
  };

  form.addEventListener('input',debouncedSave);

  saveBtn.addEventListener('click',()=>{
    const defName=document.getElementById('char-name').value||'未命名档案';
    const name=prompt('为存档命名：',defName);
    if(name&&name.trim()){
      const profiles=getProfiles();profiles[name.trim()]=getFormData();
      saveProfiles(profiles);refreshDropdown();profileSelect.value=name.trim();
      notify('存档「'+name.trim()+'」已保存');
    }
  });

  loadBtn.addEventListener('click',()=>{
    const name=profileSelect.value;
    if(name){const profiles=getProfiles();if(profiles[name]){populateForm(profiles[name]);debouncedSave();notify('已读取存档「'+name+'」');}}
    else notify('请先选择存档','#e05555');
  });

  deleteBtn.addEventListener('click',()=>{
    const name=profileSelect.value;
    if(name&&confirm('确定删除存档「'+name+'」？此操作不可恢复。')){
      const profiles=getProfiles();delete profiles[name];saveProfiles(profiles);refreshDropdown();notify('存档「'+name+'」已删除','#c4a24a');
    }
  });

  exportBtn.addEventListener('click',()=>{
    const profiles=getProfiles();if(Object.keys(profiles).length===0)return;
    const blob=new Blob([JSON.stringify(profiles,null,2)],{type:'application/json'});
    const a=document.createElement('a');a.href=URL.createObjectURL(blob);
    a.download='nls-profiles-'+Date.now()+'.json';document.body.appendChild(a);a.click();document.body.removeChild(a);notify('存档已导出');
  });

  importBtn.addEventListener('click',()=>importFile.click());
  importFile.addEventListener('change',(e)=>{
    const file=e.target.files[0];if(!file)return;
    const reader=new FileReader();
    reader.onload=(ev)=>{
      try{
        const imported=JSON.parse(ev.target.result);
        let curr=getProfiles(),added=0,skipped=0;
        for(const n in imported){
          if(curr[n]){if(confirm('已存在同名存档「'+n+'」，覆盖？')){curr[n]=imported[n];added++;}else skipped++;}
          else{curr[n]=imported[n];added++;}
        }
        saveProfiles(curr);refreshDropdown();notify('导入 '+added+' 个，跳过 '+skipped+' 个');
      }catch(err){notify('导入失败：文件格式无效','#e05555');}
      importFile.value='';
    };
    reader.readAsText(file);
  });

  resetBtn.addEventListener('click',()=>{
    if(confirm('确定清空所有填写内容？')){form.reset();debouncedSave();notify('表单已清空','#c4a24a');}
  });

  generateBtn.addEventListener('click',()=>{
    autoSave();generateBtn.disabled=true;
    generateBtn.innerHTML='<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" style="width:15px;height:15px;animation:spin 1s linear infinite"><circle cx="12" cy="12" r="10"/><path d="M12 6v6l4 2"/></svg><span>提交中...</span>';
    notification.style.opacity='0';

    const data=getFormData();
    const fmt=(v,placeholder='[未填写，请AI自由发挥补全]')=>{const t=(v||'').trim();return t?t:placeholder;};

    const yaml=
`<init_data>
\`\`\`yaml
# AI任务：解析以下创角档案，初始化所有变量并生成开场白。
# 缺失信息请AI基于关键词进行富有想象力的扩写补全。

创角档案:
  基本信息:
    姓名: "${fmt(data['char-name'],'未知 (请AI赋予日式姓名)')}"
    年龄: "${fmt(data['char-age'],'18岁')}"
    性格: "${fmt(data['char-personality'],'请AI根据背景自由发挥')}"
    背景: "${fmt(data['char-bg'],'汐望高中学生，独居涩谷公寓')}"
    主播名: \"${fmt(data['streamer-name'],'请AI自由发挥')}\"
  外貌与身材:
    外貌关键词: "${fmt(data['appearance-keywords'])}"
    身体关键词: "${fmt(data['body-keywords'])}"
    性器官关键词: \"${fmt(data['genital-keywords'])}\"
  额外备注: "${fmt(data['extra-notes'],'无')}"
\`\`\`
</init_data>`;

    setTimeout(()=>{
      try{
        const inputArea=window.parent.document.querySelector('#send_textarea');
        const sendBtn=window.parent.document.querySelector('#send_but');
        if(inputArea&&sendBtn){
          inputArea.value=yaml;inputArea.dispatchEvent(new Event('input',{bubbles:true}));
          sendBtn.click();notify('档案已提交！等待AI回应...');
          generateBtn.innerHTML='<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" style="width:15px;height:15px"><polyline points="20 6 9 17 4 12"/></svg><span>已提交</span>';
        }else throw new Error('UI not found');
      }catch(err){
        console.error(err);generateBtn.innerHTML='<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" style="width:15px;height:15px"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg><span>发送失败</span>';
        navigator.clipboard.writeText(yaml).then(()=>{notify('自动发送失败！已复制到剪贴板，请手动粘贴发送。','#e05555');});
      }
      setTimeout(()=>{generateBtn.disabled=false;generateBtn.innerHTML=initialBtnHTML;},5000);
    },500);
  });

  // Init
  refreshDropdown();
  const auto=localStorage.getItem(AUTOSAVE_KEY);
  if(auto)populateForm(JSON.parse(auto));
});
</script>
