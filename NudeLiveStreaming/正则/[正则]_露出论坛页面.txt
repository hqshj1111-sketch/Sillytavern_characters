<script>
(function(){
  var raw = document.getElementById('__forum_data');
  if (!raw) return;
  var text = raw.textContent.trim();
  if (!text) return;

  var posts = [];
  var blocks = text.split(/\n(?=\[)/);
  
  blocks.forEach(function(block){
    block = block.trim();
    if (!block) return;
    var lines = block.split('\n');
    
    // Parse header: [Section] Title
    var headerMatch = lines[0].match(/^\[(.+?)\]\s*(.+)$/);
    if (!headerMatch) return;
    var section = headerMatch[1].trim();
    var title = headerMatch[2].trim();
    
    var author = '', date = '', views = '', tags = '';
    var bodyLines = [];
    var replies = [];
    var inBody = false, inReplies = false, currentReply = '';
    
    for (var i = 1; i < lines.length; i++) {
      var line = lines[i];
      
      // Meta line: 作者: xxx | date | 浏览量: N
      if (line.match(/^作者:/)) {
        var m = line.match(/^作者:\s*(.+?)\s*\|\s*(.+?)\s*\|\s*浏览量:\s*(\d+)/);
        if (m) { author = m[1].trim(); date = m[2].trim(); views = m[3].trim(); }
        continue;
      }
      
      // Tags line
      if (line.match(/^标签:/)) {
        var tm = line.match(/^标签:\s*(.+)/);
        if (tm) tags = tm[1].trim();
        continue;
      }
      
      // Reply separator
      if (line.match(/^---回复---$/)) {
        inBody = false;
        inReplies = true;
        continue;
      }
      
      // Between replies: ---
      if (inReplies && line.match(/^---$/)) {
        if (currentReply) replies.push(currentReply.trim());
        currentReply = '';
        continue;
      }
      
      // Empty line transitions
      if (line.trim() === '') {
        if (inBody && bodyLines.length > 0) inBody = false;
        continue;
      }
      
      if (!inBody && !inReplies && line.match(/^作者:/)) {
        // Already handled above
        continue;
      }
      
      if (!inBody && !inReplies) {
        inBody = true;
      }
      
      if (inBody) {
        bodyLines.push(line);
      } else if (inReplies) {
        if (currentReply) currentReply += '\n' + line;
        else currentReply = line;
      }
    }
    if (currentReply) replies.push(currentReply.trim());
    
    var body = bodyLines.join('\n').trim();
    if (!body) return;
    
    posts.push({ section: section, title: title, author: author, date: date, views: views, tags: tags, body: body, replies: replies });
  });

  var container = document.getElementById('posts');
  if (posts.length === 0) {
    container.innerHTML = '<div class="no-posts">暂时没有帖子。输入"逛论坛"查看最新讨论。</div>';
    return;
  }

  var sectionClass = {
    '露出日常': '露出日常',
    '直播心得': '直播心得',
    '技术交流': '技术交流',
    '邀约': '邀约',
    '情感': '情感'
  };

  posts.forEach(function(p, idx){
    var sc = sectionClass[p.section] || '';
    var tagsHtml = '';
    if (p.tags) {
      var tarr = p.tags.split(/\s*#\s*/).filter(function(t){ return t.length>0; });
      if (tarr.length > 0) {
        tagsHtml = '<div class="post-tags">' + tarr.map(function(t){ return '<span class="post-tag">#' + t + '</span>'; }).join('') + '</div>';
      }
    }
    
    var repliesHtml = '';
    if (p.replies.length > 0) {
      var rCount = p.replies.length;
      var rid = 'replies-' + idx;
      var replyItems = p.replies.map(function(r){
        var parts = r.split('\n');
        var firstLine = parts[0];
        var colonIdx = firstLine.indexOf(':');
        var rAuthor = '', rContent = '';
        if (colonIdx > 0) {
          rAuthor = firstLine.substring(0, colonIdx).trim();
          rContent = firstLine.substring(colonIdx + 1).trim();
          if (parts.length > 1) rContent += '\n' + parts.slice(1).join('\n');
        } else {
          rContent = r;
        }
        return '<div class="reply-item"><div class="reply-author">' + rAuthor + '</div><div class="reply-content">' + rContent + '</div></div>';
      }).join('');

      repliesHtml = 
        '<div class="post-replies">' +
          '<div class="replies-toggle" onclick="var r=document.getElementById(\''+rid+'\');var t=this;if(r.style.display==\'block\'){r.style.display=\'none\';t.classList.remove(\'open\')}else{r.style.display=\'block\';t.classList.add(\'open\')}">' +
            '<span class="arrow">▶</span> ' + rCount + ' 条回复' +
          '</div>' +
          '<div class="reply-list" id="' + rid + '">' + replyItems + '</div>' +
        '</div>';
    }

    container.insertAdjacentHTML('beforeend',
      '<div class="post-card">' +
        '<div class="post-header">' +
          '<span class="post-section sec-' + sc + '">[' + p.section + ']</span>' +
          '<div class="post-title">' + p.title + '</div>' +
          '<div class="post-meta">' +
            '<span>作者: ' + p.author + '</span>' +
            '<span>' + p.date + '</span>' +
            '<span>👁 ' + p.views + '</span>' +
          '</div>' +
          tagsHtml +
        '</div>' +
        '<div class="post-body">' + p.body + '</div>' +
        repliesHtml +
      '</div>'
    );
  });
})();
</script>
