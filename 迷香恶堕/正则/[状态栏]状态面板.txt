// === 工具安全取值函数 ===
var safeStr = function(v, fallback) { return (v !== undefined && v !== null && v !== '' && v !== '待设定') ? String(v) : (fallback || '---'); };
var safeNum = function(v) { return (v !== undefined && v !== null && !isNaN(Number(v))) ? Number(v) : 0; };
var pct = function(cur, max) { if (!max || max <= 0) return 0; return Math.min(100, Math.round((cur / max) * 100)); };

function parseActionOptionsText(rawText) {
  if (!rawText) return null;
  var lines = rawText.split('\n')
    .map(function(l) { return l.trim(); })
    .filter(function(l) { return l.length > 0; });

  var parsed = [];
  lines.forEach(function(line) {
    var clean = line.replace(/^(\d+[\.、\s\-\)]+|[-*•]\s+|[\[【(（]\d+[\]】)）]\s*|[①②③④⑤⑥⑦⑧⑨⑩]\s*)/, '').trim();
    if (clean) {
      parsed.push({
        text: clean,
        cmd: clean
      });
    }
  });
  return parsed.length > 0 ? parsed.slice(0, 4) : null;
}

function getRawActionsFromDOM() {
  try {
    var hostMes = (document.closest ? document.closest('.mes_text') : null);
    if (!hostMes && window.parent && window.parent.document) {
      var allMes = window.parent.document.querySelectorAll('.mes_text');
      if (allMes.length > 0) hostMes = allMes[allMes.length - 1];
    }
    if (hostMes) {
      var actEl = hostMes.querySelector('.mxe-raw-actions');
      if (actEl && actEl.textContent.trim()) {
        return parseActionOptionsText(actEl.textContent.trim());
      }
      var rawMatch = hostMes.textContent.match(/<行动选项>\s*([\s\S]*?)\s*<\/行动选项>/i);
      if (rawMatch && rawMatch[1].trim()) {
        return parseActionOptionsText(rawMatch[1].trim());
      }
    }
  } catch(e) {}
  return null;
}

// === 视角切换弹窗逻辑 ===
function openPerspectiveModal() {
  var modal = document.getElementById('perspective-modal');
  if (!modal) return;
  var curPerspective = (window._mxeCurrentData && window._mxeCurrentData.视角) || '怪人主角';
  selectModalPerspective(curPerspective);
  modal.classList.add('show');
}

function closePerspectiveModal() {
  var modal = document.getElementById('perspective-modal');
  if (modal) modal.classList.remove('show');
}

function switchTab(tabId) {
  window._mxeActiveTab = tabId;
  document.querySelectorAll('.tab-btn').forEach(function(btn) { btn.classList.remove('active'); });
  document.querySelectorAll('.tab-panel').forEach(function(panel) { panel.classList.remove('active'); });

  if (event && event.currentTarget) event.currentTarget.classList.add('active');
  var target = document.getElementById(tabId);
  if (target) target.classList.add('active');
}

}

// === MVU 变量框架集成 ===
function getMvuInstance() {
  try { return window.Mvu || (window.parent && window.parent.Mvu) || (window.top && window.top.Mvu) || null; }
  catch(e) { return null; }
}

const init = function() {
  var Mvu = getMvuInstance();
  if (Mvu && Mvu.events) {
    document.addEventListener(Mvu.events.VARIABLE_UPDATE_ENDED, function(e) {
      if (e.detail && e.detail.stat_data) {
        var domAct = getRawActionsFromDOM();
        if (domAct) window._mxeDynamicActions = domAct;
        renderPanelWithData(e.detail.stat_data);
      }
    });
    (async function() {
      try {
        var data;
        if (typeof getChatMessages !== 'undefined' && typeof getCurrentMessageId !== 'undefined') {
          var msgs = await getChatMessages(getCurrentMessageId());
          data = msgs[0]?.data?.stat_data || Mvu.getMvuData({type:'message', message_id:-1})?.stat_data;
          var rawMsg = msgs[0]?.mes || msgs[0]?.message || '';
          var actMatch = rawMsg.match(/<行动选项>\s*([\s\S]*?)\s*<\/行动选项>/i);
          if (actMatch && actMatch[1].trim()) {
            window._mxeDynamicActions = parseActionOptionsText(actMatch[1].trim());
          }
        } else {
          data = Mvu.getMvuData({type:'message', message_id:-1})?.stat_data;
        }
        var domAct = getRawActionsFromDOM();
        if (domAct) window._mxeDynamicActions = domAct;
        if (data) { renderPanelWithData(data); }
      } catch(e) {}
    })();
  } else {
    var domAct = getRawActionsFromDOM();
    if (domAct) {
      window._mxeDynamicActions = domAct;
      if (window._mxeCurrentData) renderPanelWithData(window._mxeCurrentData);
    }
    setTimeout(init, 500);
  }
};
