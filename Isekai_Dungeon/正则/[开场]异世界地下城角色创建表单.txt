<script>
document.addEventListener('DOMContentLoaded',function(){
  var form=document.getElementById('creator-form');
  var generateBtn=document.getElementById('generate-btn');
  var resetBtn=document.getElementById('reset-btn');
  var notification=document.getElementById('notification');
  var allInputs=Array.from(form.querySelectorAll('input[type=text], textarea, select'));
  var initialBtnHTML=generateBtn.innerHTML;
  var AUTOSAVE_KEY='idf_charCreate_autosave_v1';
  var saveTimeout;

  // City data
  var CITY_DATA={
    '龙脊联邦':['霜脊','烬喉','灼鳞','雾骸','裂渊'],
    '霜境王国':['黑脉镇','长风驿','霜木垒','铁砧堡','冰湾港'],
    '密林双廷':['影渊廷','藤崖渡','林曦塞','月泉台','叶冕庭'],
    '原野同盟':['沙角营','河鬃渡','蹄风港','百牙原','霜爪垒'],
    '金沙帝国':['金帆港','砂碑关','烈风驿','日轮都','炎宴城'],
    '潮汐联邦':['蓝渊礁','焰岛','珊链屿','扬帆湾','汐音屿']
  };

  var homelandSelect=document.getElementById('mc-homeland');
  var homecitySelect=document.getElementById('mc-homecity');

  Object.keys(CITY_DATA).forEach(function(cn){
    var opt=document.createElement('option');opt.value=cn;opt.textContent=cn;homelandSelect.appendChild(opt);
  });

  var notify=function(msg,color){
    color=color||'#5a9a5a';notification.textContent=msg;notification.style.color=color;notification.style.opacity='1';
    clearTimeout(notification._tid);notification._tid=setTimeout(function(){notification.style.opacity='0'},3500);
  };

  var getFormData=function(){var d={};allInputs.forEach(function(inp){if(inp.id)d[inp.id]=inp.value});return d};

  var populateForm=function(data){if(!data)return;allInputs.forEach(function(inp){var k=inp.id;if(data[k]!==undefined){if(inp===homelandSelect){inp.value=data[k];inp.dispatchEvent(new Event('change'))}else{inp.value=data[k]}}else if(inp.tagName!=='SELECT')inp.value=''})};

  var autoSave=function(){localStorage.setItem(AUTOSAVE_KEY,JSON.stringify(getFormData()))};
  var debouncedSave=function(){clearTimeout(saveTimeout);saveTimeout=setTimeout(autoSave,500)};

  // Gender toggle for genital label
  document.getElementById('mc-gender').addEventListener('change',function(){
    var isMale=this.value==='男';
    var lbl=document.getElementById('lbl-genital');
    var ta=document.getElementById('mc-genital');
    lbl.textContent='性器官特征描述'+(isMale?' (阴茎形态/后穴形态/胸部形态)':' (小穴形态/后穴形态/胸部形态/乳头形态)');
    if(isMale)ta.placeholder=ta.placeholder.indexOf('阴茎')===-1?'阴茎中等尺寸颜色偏浅，后穴未经开发紧致粉嫩，胸部平坦结实...':ta.placeholder;
    else ta.placeholder=ta.placeholder.indexOf('小穴')===-1?'小穴紧致敏感内侧浅粉色，后穴未经开发，胸部柔软微微隆起，乳头小巧浅褐色...':ta.placeholder;
    debouncedSave();
  });

  homelandSelect.addEventListener('change',function(){
    homecitySelect.innerHTML='';
    var cities=CITY_DATA[this.value]||[];
    if(cities.length){
      var opt=document.createElement('option');opt.value='';opt.textContent='选择城市...';homecitySelect.appendChild(opt);
      cities.forEach(function(ct){
        var o=document.createElement('option');o.value=ct;o.textContent=ct;homecitySelect.appendChild(o);
      });
    }else{
      var opt=document.createElement('option');opt.value='';opt.textContent='先选择国家...';homecitySelect.appendChild(opt);
    }
    debouncedSave();
  });

  form.addEventListener('input',debouncedSave);

  resetBtn.addEventListener('click',function(){
    if(confirm('确定清空所有填写内容？')){form.reset();debouncedSave();notify('表单已清空','#c4a24a')}
  });

  generateBtn.addEventListener('click',function(){
    autoSave();generateBtn.disabled=true;
    generateBtn.innerHTML='⚔️ 提交中...';
    notification.style.opacity='0';

    var data=getFormData();
    var fmt=function(v,ph){var t=(v||'').trim();return t||(ph||'[未填写，请AI自由发挥补全]')};

    var yaml=
'<init_data>\n'+
'```yaml\n'+
'# AI任务：解析以下创角档案，初始化主角的全部变量（基本信息、相貌特征、身材细节、性器官特征），\n'+
'# 并根据档案内容生成开场的冒险者公会登记场景。缺失信息请AI基于关键词进行富有想象力的扩写补全。\n'+
'\n'+
'创角档案:\n'+
'  基本信息:\n'+
'    姓名: "'+fmt(data['mc-name'],'未知 (请AI赋予奇幻风格姓名)')+'"\n'+
'    性别: "'+fmt(data['mc-gender'],'男')+'"\n'+
'    年龄: "'+fmt(data['mc-age'],'18岁')+'"\n'+
'    种族: "'+fmt(data['mc-race'],'人类')+'"\n'+
'    职业: "'+fmt(data['mc-job'],'剑士')+'"\n'+
'    身份: "'+fmt(data['mc-identity'],'冒险者')+'"\n'+
'    性格: "'+fmt(data['mc-personality'],'勇敢果断但有时冲动，对同伴忠诚')+'"\n'+
'    故乡国家: "'+fmt(data['mc-homeland'],'龙脊联邦')+'"\n'+
'    故乡城市: "'+fmt(data['mc-homecity'],'烬喉')+'"\n'+
'    背景: "'+fmt(data['mc-background'],'普通冒险者出身')+'"\n'+
'  外貌与身材:\n'+
'    外貌关键词: "'+fmt(data['mc-appearance'])+'"\n'+
'    身体关键词: "'+fmt(data['mc-body'])+'"\n'+
'    性器官关键词: "'+fmt(data['mc-genital'],'[未填写，请AI根据性别合理设定]')+'"\n'+
'```\n'+
'</init_data>';

    setTimeout(function(){
      try{
        var inputArea=window.parent.document.querySelector('#send_textarea');
        var sendBtn=window.parent.document.querySelector('#send_but');
        if(inputArea&&sendBtn){
          inputArea.value=yaml;inputArea.dispatchEvent(new Event('input',{bubbles:true}));
          sendBtn.click();notify('档案已提交！等待AI回应...');
          generateBtn.innerHTML='✅ 已提交';
        }else throw new Error('UI not found');
      }catch(err){
        console.error(err);generateBtn.innerHTML='发送失败';
        navigator.clipboard.writeText(yaml).then(function(){notify('自动发送失败！已复制到剪贴板，请手动粘贴发送。','#c0392b')});
      }
      setTimeout(function(){generateBtn.disabled=false;generateBtn.innerHTML=initialBtnHTML},5000);
    },500);
  });

  // Init
  var auto=localStorage.getItem(AUTOSAVE_KEY);
  if(auto)populateForm(JSON.parse(auto));
  document.getElementById('mc-gender').dispatchEvent(new Event('change'));
});
</script>
</body>
